===== apps/dashboards/services.py =====
"""dashboards.services: everything a dashboard page shows.

``get_dashboard_context()`` is the only entry point. The views stay thin and hand
the result to a template. Pieces:

* shared by every role: greeting name, verse of the day (in the reader's Bible
  version), "continue reading" (the person's Bible bookmark), quick links (taken from ``NAV_ITEMS``, so a menu
  change is picked up here too), the faith widgets (streaks and today's goals: read-only
  numbers from ``apps.faith.overview``) and the "coming soon" cards;
* leaders only (super-admin, admin, coordinator): counts and charts for what they
  manage. Attendance and tithes numbers come from *uploaded* sheets (tithes: finalized
  reports) and use the frozen values stored on them, never a recount.

Charts are drawn by the templates from plain ``Bar`` rows (label, text, percent),
so there is no JavaScript and the colors follow the theme tokens.

Brick 1 of the dashboard revamp adds the *data layer* next to the old pieces (nothing old was removed):
``resolve_extension`` (the admin's ``?extension=`` filter) and the stats functions ``attendance_stats``,
``people_stats``, ``extension_stats``, ``giving_stats``, ``prayer_summary`` and ``upcoming_events``. Each returns
plain dicts and lists (no templates, no chart geometry: that is ``charts.py``). Scoping still happens ONLY in
``_scope_people``, ``_scope_attendance`` and ``_scope_tithes``; prayer and events keep their own single scoping
helpers (``inbox_for`` and ``visible_events``).

Brick 2 adds what the leader pages draw: ``tiles`` (four headline numbers), ``charts`` (every chart's geometry from
``charts.py`` plus its text, ready for the ``_chart_*`` partials) and ``events_url``. They are built from the Brick 1
stats, so they follow the same scoping and the same ``?extension=`` filter and add no query of their own.

Privacy: tithe amounts are shown only to leaders, only here, and are never written
to ``audit_log`` or ``notification.data``.
"""

def get_dashboard_context(user, version_code, path, extension=None, start=None, end=None):
    """Return the ``dashboard`` dict a role template renders.

    ``version_code`` is the Bible version cookie (or None) and ``path`` is the
    current URL path, used to mark quick links. ``leader`` is None for members.
    ``streaks`` and ``goals_today`` are None for people who cannot use goals (the templates then draw nothing).

    Optional (the views do not pass them yet, so every old call behaves as before):

    * ``extension``: the raw ``?extension=`` text. Only admin and super-admin may use it; a bad value means
      "All", a coordinator always gets their own extension (see ``resolve_extension``).
    * ``start`` / ``end``: dates (Manila). With neither, the stats cover the last ``TREND_SHEETS`` gathering dates.

    New keys (all present for every role; None where the role may not see the data):
    ``filter``, ``period``, ``attendance_stats``, ``people_stats``, ``extension_stats``, ``giving_stats``,
    ``prayer_summary`` and ``upcoming_events`` (a list, for every role).

    Brick 2 keys: ``tiles`` (list of ``{"label", "value", "hint"}``), ``charts`` (see ``leader_charts``; both None
    for a member) and ``events_url`` (the calendar page, "" if no events route resolves). The old ``leader``
    block is unchanged and still ignores the filter; the leader pages no longer draw it.
    """
    today = timezone.localdate()
    is_leader = user.role in LEADER_ROLES
    chosen = resolve_extension(user, extension) if is_leader else None
    extension_id = chosen.pk if chosen else None

    attendance = attendance_stats(user, extension_id, start, end)
    # The period every other piece uses. By default it is exactly the dates the attendance trend shows.
    period_start, period_end = start, end
    if attendance and start is None and end is None and attendance["period"]:
        period_start = attendance["period"]["start"]
        period_end = max(attendance["period"]["end"], today)

    people = people_stats(user, extension_id, period_start, period_end)
    extensions = extension_stats(user, period_start, period_end)
    giving = giving_stats(user, extension_id, period_start, period_end)
    period_given = start is not None or end is not None

    return {
        "greeting_name": title_case(user.first_name),
        "verse": _verse_of_the_day(version_code),
        "continue_reading": _continue_reading(user),
        "bible_url": _safe_reverse("bible:reader"),
        "start_url": _bible_chapter_url("Gen", 1),
        "quick_links": _quick_links(user, path),
        "streaks": _streaks(user, today),
        "goals_today": _goals_today(user, today),
        "coming_soon": COMING_SOON,
        "leader": _leader_overview(user) if is_leader else None,
        "filter": _filter_info(user, chosen),
        "period": {"start": period_start, "end": period_end} if (period_start or period_end) else None,
        "attendance_stats": attendance,
        "people_stats": people,
        "extension_stats": extensions,
        "giving_stats": giving,
        "prayer_summary": prayer_summary(user, extension_id),
        "upcoming_events": upcoming_events(user, 5),
        "events_url": _safe_reverse("events:calendar") or _safe_reverse("events:list") or _safe_reverse("events:index"),
        "tiles": dashboard_tiles(attendance, people, extensions, period_start) if is_leader else None,
        "charts": leader_charts(attendance, people, extensions, giving, period_given) if is_leader else None,
    }


# --------------------------------------------------------------- scoping
# AGENTS.md says coordinator-scoped data goes through the project's for_user(user)
# queryset helper. These three functions are the only places this app narrows
# data to a coordinator's extension. Attendance and tithes go through their own
# `for_user`; only people still filter by hand.
def _narrow(queryset, user, extension_id, field):
    """Narrow to one extension for admin and super-admin only; a coordinator is already narrowed by `for_user`.

    ``extension_id`` is the already validated id from ``resolve_extension`` (never raw request text).
    """
    if extension_id and user.role != Role.COORDINATOR:
        return queryset.filter(**{field: extension_id})
    return queryset


def _within(queryset, field, start, end):
    """Keep rows whose date ``field`` falls between ``start`` and ``end`` (either may be None = open)."""
    if start is not None:
        queryset = queryset.filter(**{f"{field}__gte": start})
    if end is not None:
        queryset = queryset.filter(**{f"{field}__lte": end})
    return queryset


def _scope_people(queryset, user, extension_id=None, start=None, end=None):
    """Limit people to the coordinator's own extension; leaders above see everyone.

    ``extension_id`` (admin and super-admin only; ignored for a coordinator) narrows to one extension.
    ``start`` / ``end`` keep people who joined (``created_at``, Manila date) in that period; leave them out
    for the plain headcount.
    """
    if user.role == Role.COORDINATOR:
        queryset = queryset.filter(extension_id=user.extension_id)
    else:
        queryset = _narrow(queryset, user, extension_id, "extension_id")
    return _within(queryset, "created_at__date", start, end)


def _scope_attendance(user, extension_id=None, start=None, end=None):
    """Uploaded sheets the person may see, through `Attendance.objects.for_user`.

    Drafts are never part of a dashboard, so only uploaded sheets are returned (a reopened
    sheet is a draft again and drops out until it is uploaded once more).
    ``extension_id`` (ignored for a coordinator) and ``start`` / ``end`` (on ``attendance_date``) narrow further.
    """
    sheets = Attendance.objects.for_user(user).filter(status=Attendance.Status.UPLOADED)
    sheets = _narrow(sheets, user, extension_id, "extension_id")
    return _within(sheets, "attendance_date", start, end)


def _scope_tithes(user, extension_id=None, start=None, end=None):
    """FINALIZED reports the person may see, through `TithesOffering.objects.for_user`.

    Same shape as `_scope_attendance`: drafts are never part of a dashboard, and a report counts only while
    its sheet is uploaded (a reopened report is a draft again, an archived or reopened sheet drops out).
    A coordinator gets their own extension only; `for_user` is the single place that decides.
    ``extension_id`` (ignored for a coordinator) and ``start`` / ``end`` (on the sheet's date) narrow further.
    """
    reports = TithesOffering.objects.for_user(user).filter(
        status=TithesOffering.Status.FINALIZED,
        attendance__status=Attendance.Status.UPLOADED,
    )
    reports = _narrow(reports, user, extension_id, "attendance__extension_id")
    return _within(reports, "attendance__attendance_date", start, end)


def resolve_extension(user, raw_value):
    """Turn the raw ``?extension=`` text into the extension to filter by, or None for "All".

    * admin and super-admin: the extension with that id, only if it exists and is not archived. Anything
      else (empty, letters, a negative or decimal number, a huge number, an unknown or archived id) is None.
    * coordinator: always their OWN extension, whatever the text says (nothing from the request is trusted).
    * anyone else: None.
    """
    if user.role == Role.COORDINATOR:
        return user.extension if user.extension_id else None
    if user.role not in FILTER_ROLES:
        return None
    text = str(raw_value or "").strip()
    # isascii first: str.isdigit() also accepts digits such as "²" that int() cannot read.
    if not (text.isascii() and text.isdigit()) or int(text) > MAX_DB_ID:
        return None
    return Extension.objects.filter(pk=int(text), archived_at__isnull=True).first()


def _filter_info(user, chosen):
    """Return what the filter bar needs: whether it shows, the picked extension and the choices."""
    enabled = user.role in FILTER_ROLES
    options = []
    if enabled:
        live = Extension.objects.filter(archived_at__isnull=True).order_by("name")
        options = [{"id": ext.pk, "name": title_case(ext.name)} for ext in live]
    return {
        "enabled": enabled,
        "extension_id": chosen.pk if chosen else None,
        "label": title_case(chosen.name) if chosen else _scope_label(user),
        "options": options,
    }


# ----------------------------------------------------------- shared pieces
def _safe_reverse(name, *args):
    """Return the URL for a route name, or "" when the route does not exist yet."""
    try:
        return reverse(name, args=args)
    except NoReverseMatch:
        return ""


def _bible_chapter_url(osis_id, chapter):
    """Return the reader URL for a chapter, or "" if the Bible routes cannot be resolved.

    Tries the named route first; otherwise builds the path under the reader's URL
    (``/bible/<book>/<chapter>/``, see the Bible section of AGENTS.md).
    """
    url = _safe_reverse("bible:chapter", osis_id, chapter)
    if url:
        return url
    base = _safe_reverse("bible:reader")
    return f"{base}{osis_id}/{chapter}/" if base else ""


def _verse_of_the_day(version_code):
    """Return today's verse in the reader's version (text, or a note when it cannot be read)."""
    version = bible_services.resolve_version(version_code)
    row = daily_verse.get_daily_verse()
    verse = daily_verse.get_daily_verse_text(row, version)
    return {
        "reference": verse.reference,
        "version": verse.version,
        "text": verse.text,
        "note": verse.note,
        "available": verse.available,
        "chapter_url": _bible_chapter_url(row.book, row.chapter),
    }


def _continue_reading(user):
    """Return the person's bookmark as ``{"label", "url"}``, or None when they have none.

    The bookmark is the only source (the reader moves it on Next and after 20 seconds on a
    chapter). With no bookmark, or one that points at a book that no longer exists, the
    template offers "Start reading" at Genesis 1 instead (``start_url`` in the context).
    """
    bookmark = bible_services.get_bookmark(user)
    book = BOOKS_BY_ID.get(bookmark.book) if bookmark else None
    if book is None:
        return None
    return {
        "label": f"{book.name} {bookmark.chapter}",
        "url": _bible_chapter_url(bookmark.book, bookmark.chapter),
    }


def _streaks(user, today):
    """Return the open and Bible-reading streaks for the streak widget, or None when the person cannot use goals."""
    if not can_use_goals(user):
        return None
    return {**faith_overview.streak_cards(user, today), "url": _safe_reverse("faith:history")}


def _goals_today(user, today):
    """Return today's goal counts and next goals for the goals widget, or None when the person cannot use goals."""
    if not can_use_goals(user):
        return None
    return {**faith_overview.today_summary(user, today), "url": _safe_reverse("faith:today")}


def _quick_links(user, path):
    """Return the person's menu entries (from NAV_ITEMS) as shortcut cards, minus Home and Django admin."""
    nav = build_nav(user.role, path)
    return [
        entry
        for section in nav["sections"]
        for entry in section["items"]
        if entry["label"] not in NAV_LABELS_NOT_QUICK_LINKS
    ]

def _leader_overview(user):
    """Return the counts and chart rows for a super-admin, admin or coordinator."""
    people = _scope_people(User.objects.all(), user)
    if user.role != Role.SUPER_ADMIN:
        people = people.exclude(role=Role.SUPER_ADMIN)  # only the super-admin sees super-admin rows

    role_counts = dict(
        people.exclude(status=Status.ARCHIVED).values_list("role").annotate(n=Count("id")).order_by()
    )
    status_counts = dict(people.values_list("status").annotate(n=Count("id")).order_by())

    attendance_rows = _attendance_trend(user)
    dates = [row["attendance_date"] for row in attendance_rows]
    tithe_amounts = _tithe_totals(user, dates)

    is_coordinator = user.role == Role.COORDINATOR
    extension_total = 0 if is_coordinator else Extension.objects.filter(archived_at__isnull=True).count()
    extension_bars = [] if is_coordinator else _people_per_extension()

    return {
        "scope_label": _scope_label(user),
        "tiles": _tiles(status_counts, extension_total, attendance_rows, is_coordinator),
        "role_bars": _bars([(label, role_counts[value]) for value, label in Role.choices if role_counts.get(value)]),
        "status_bars": _bars(
            [(label, status_counts[value]) for value, label in Status.choices if status_counts.get(value)]
        ),
        "show_extensions": not is_coordinator,
        "extension_bars": extension_bars,
        "extension_more": max(0, extension_total - len(extension_bars)),
        "attendance_bars": _bars([(_day_label(r["attendance_date"]), r["present"]) for r in attendance_rows]),
        "first_timer_bars": _bars([(_day_label(r["attendance_date"]), r["first"]) for r in attendance_rows]),
        "tithe_bars": _bars(
            [(_day_label(day), tithe_amounts.get(day, {}).get("tithes", Decimal("0"))) for day in dates],
            display=_peso,
            short=_compact,
        ),
        "offering_bars": _bars(
            [(_day_label(day), tithe_amounts.get(day, {}).get("offering", Decimal("0"))) for day in dates],
            display=_peso,
            short=_compact,
        ),
    }


def _scope_label(user):
    """Say whose numbers these are: one extension for a coordinator, otherwise everyone."""
    if user.role == Role.COORDINATOR and user.extension_id:
        return title_case(user.extension.name)
    return "All extensions"


def _tiles(status_counts, extension_total, attendance_rows, is_coordinator):
    """Return the small headline numbers at the top of a leader dashboard."""
    tiles = [{"label": "Active people", "value": status_counts.get(Status.ACTIVE.value, 0), "hint": "Activated accounts"}]
    if not is_coordinator:
        tiles.append({"label": "Extensions", "value": extension_total, "hint": "Not archived"})
    if attendance_rows:
        latest = attendance_rows[-1]
        tiles.append(
            {"label": "Latest attendance", "value": latest["present"], "hint": _day_label(latest["attendance_date"])}
        )
    else:
        tiles.append({"label": "Latest attendance", "value": "–", "hint": "No uploaded sheets yet"})
    return tiles


def _attendance_trend(user):
    """Return the last few gathering dates (oldest first) with frozen present and first-timer totals.

    Only uploaded sheets count (drafts are not final, archived ones are gone). Sheets
    on the same date, for example from several extensions, are added together.
    """
    sheets = _scope_attendance(user)
    rows = list(
        sheets.values("attendance_date")
        .annotate(present=Coalesce(Sum("total_present"), 0), first=Coalesce(Sum("first_timers_count"), 0))
        .order_by("-attendance_date")[:TREND_SHEETS]
    )
    rows.reverse()
    return rows


def _tithe_totals(user, dates):
    """Return {date: {"tithes", "offering"}} of FROZEN values for finalized reports on those dates.

    Never a recount: `tithes_total` and `offering_amount` are read as stored. Reports on the same date
    (several services, several extensions) are added together.
    """
    if not dates:
        return {}
    rows = (
        _scope_tithes(user).filter(attendance__attendance_date__in=dates)
        .values("attendance__attendance_date")
        .annotate(tithes=Sum("tithes_total"), offering=Sum("offering_amount")).order_by()
    )
    return {
        row["attendance__attendance_date"]: {"tithes": row["tithes"], "offering": row["offering"]}
        for row in rows
    }


def _people_per_extension():
    """Return bars of the biggest extensions by people who are not archived (super-admin and admin only)."""
    rows = (
        Extension.objects.filter(archived_at__isnull=True)
        .annotate(n=Count("members", filter=~Q(members__status=Status.ARCHIVED.value)))
        .order_by("-n", "name")[:EXTENSION_BAR_LIMIT]
    )
    return _bars([(title_case(extension.name), extension.n) for extension in rows])


# -------------------------------------------------------- stats (Brick 1)
# Plain dicts and lists, one function per dashboard card family. None of them renders anything, checks a
# request or writes to the database. Money stays Decimal. Leader-only functions return None for a member.
def _gathering_dates(user, extension_id=None, start=None, end=None):
    """Return the gathering dates to report on, oldest first.

    With ``start`` or ``end``: every date that has an uploaded sheet in that period. With neither: the last
    ``TREND_SHEETS`` dates. Several sheets on one date (other extensions, other services) are one date.
    """
    sheets = _scope_attendance(user, extension_id, start, end)
    dates = sheets.order_by("-attendance_date").values_list("attendance_date", flat=True).distinct()
    if start is None and end is None:
        dates = dates[:TREND_SHEETS]
    return sorted(dates)


def _merge_summaries(summaries):
    """Add several frozen ``Attendance.summary`` JSON values together (same shape as one summary).

    A missing summary (None) or a missing key counts as 0, like ``summary_table`` does.
    """
    merged = {}
    for summary in summaries:
        for group, genders in (summary or {}).items():
            for gender, networks in (genders or {}).items():
                cell = merged.setdefault(group, {}).setdefault(gender, {})
                for network, number in (networks or {}).items():
                    cell[network] = cell.get(network, 0) + int(number or 0)
    return merged


def _gender_by_network(summary):
    """Return men and women by age network from one (merged) summary: first timers, guests and members together.

    Shape: ``{"columns": [{"code", "label"}], "rows": [{"gender", "label", "values", "total"}], "total": n}``.
    The rows are ready for ``charts.column_groups`` (one group per column, one value per row).
    """
    table = summary_table(summary)
    values_by_gender = {}
    for group in table["groups"]:
        for row in group["rows"]:
            current = values_by_gender.get(row["gender"], [0] * len(table["columns"]))
            values_by_gender[row["gender"]] = [a + b for a, b in zip(current, row["cells"])]
    rows = [
        {"gender": gender, "label": GENDER_ROW_LABELS[gender], "values": values, "total": sum(values)}
        for gender, values in values_by_gender.items()
    ]
    return {"columns": table["columns"], "rows": rows, "total": table["total"]}


def attendance_stats(user, extension_id=None, start=None, end=None):
    """Return the attendance numbers of the uploaded sheets in scope, or None for a member.

    Everything comes from FROZEN values (``total_present``, ``first_timers_count``, ``summary``), never a
    recount, and sheets on the same date are added together. Keys:

    * ``points``: one per gathering date, oldest first: ``date``, ``label``, ``present``, ``first_timers``,
      ``men``, ``women``. ``labels`` and ``values`` repeat label and present for ``charts.line_chart``.
    * ``count``, ``total_present``, ``average`` (one decimal, 0 with no dates), ``first_timers`` (sum),
      ``highest`` and ``lowest`` (a point; the most recent wins a tie), ``latest`` (the newest point).
    * ``network``: men and women by age network over ALL dates; ``network_latest``: the same for the
      newest date only (``_gender_by_network`` shape); ``split``: ``{"men", "women"}`` over all dates.
    * ``period``: ``{"start", "end"}`` of the dates covered, or None with no uploaded sheets.
    """
    if user.role not in LEADER_ROLES:
        return None
    dates = _gathering_dates(user, extension_id, start, end)
    sheets = _scope_attendance(user, extension_id).filter(attendance_date__in=dates).values(
        "attendance_date", "total_present", "first_timers_count", "summary"
    )
    by_date = {day: {"present": 0, "first": 0, "summaries": []} for day in dates}
    for sheet in sheets:
        row = by_date[sheet["attendance_date"]]
        row["present"] += sheet["total_present"] or 0
        row["first"] += sheet["first_timers_count"] or 0
        row["summaries"].append(sheet["summary"])

    points, per_date_network = [], {}
    for day in dates:
        row = by_date[day]
        network = _gender_by_network(_merge_summaries(row["summaries"]))
        per_date_network[day] = network
        totals = {r["gender"]: r["total"] for r in network["rows"]}
        points.append({
            "date": day, "label": _day_label(day), "present": row["present"], "first_timers": row["first"],
            "men": totals.get("male", 0), "women": totals.get("female", 0),
        })

    every_summary = [s for day in dates for s in by_date[day]["summaries"]]
    total_present = sum(p["present"] for p in points)
    return {
        "points": points,
        "labels": [p["label"] for p in points],
        "values": [p["present"] for p in points],
        "count": len(points),
        "total_present": total_present,
        "average": round(total_present / len(points), 1) if points else 0,
        "first_timers": sum(p["first_timers"] for p in points),
        "highest": max(points, key=lambda p: (p["present"], p["date"]), default=None),
        "lowest": min(points, key=lambda p: (p["present"], -p["date"].toordinal()), default=None),
        "latest": points[-1] if points else None,
        "network": _gender_by_network(_merge_summaries(every_summary)),
        "network_latest": per_date_network[dates[-1]] if dates else _gender_by_network({}),
        "split": {"men": sum(p["men"] for p in points), "women": sum(p["women"] for p in points)},
        "period": {"start": dates[0], "end": dates[-1]} if dates else None,
    }


def people_stats(user, extension_id=None, start=None, end=None):
    """Return the people counts for a leader (None for a member).

    Same rules as the old overview: only the super-admin sees super-admin rows, the roster (``roles``) skips
    archived people and ``statuses`` includes them. Keys: ``total`` (not archived), ``active``, ``roles`` and
    ``statuses`` (lists of ``{"value", "label", "count"}``, zero rows left out) and ``new_in_period``: how many
    people who are not archived joined between ``start`` and ``end`` (None when no period is given).
    """
    if user.role not in LEADER_ROLES:
        return None
    people = _scope_people(User.objects.all(), user, extension_id)
    if user.role != Role.SUPER_ADMIN:
        people = people.exclude(role=Role.SUPER_ADMIN)
    live = people.exclude(status=Status.ARCHIVED)
    role_counts = dict(live.values_list("role").annotate(n=Count("id")).order_by())
    status_counts = dict(people.values_list("status").annotate(n=Count("id")).order_by())
    new_in_period = None
    if start is not None or end is not None:
        new_in_period = _scope_people(live, user, extension_id, start, end).count()
    return {
        "total": sum(role_counts.values()),
        "active": status_counts.get(Status.ACTIVE.value, 0),
        "roles": [
            {"value": value, "label": label, "count": role_counts[value]}
            for value, label in Role.choices if role_counts.get(value)
        ],
        "statuses": [
            {"value": value, "label": label, "count": status_counts[value]}
            for value, label in Status.choices if status_counts.get(value)
        ],
        "new_in_period": new_in_period,
    }


def extension_stats(user, start=None, end=None):
    """Return the extension comparison for admin and super-admin (None for everyone else).

    Never narrowed by the extension filter: it is the comparison ACROSS extensions. Keys: ``total`` (not
    archived), ``with_coordinator`` and ``without_coordinator`` (a coordinator counts only if they can sign
    in, the same rule as prayer), ``silent`` (extensions with no uploaded sheet in the last ``SILENT_DAYS``
    days, as ``{"id", "name"}``), ``silent_days``, ``attendance_per_extension`` and ``people_per_extension``
    (top ``EXTENSION_BAR_LIMIT`` as ``{"name", "count"}``) and ``people_more`` (extensions not listed there).
    """
    if user.role not in FILTER_ROLES:
        return None
    today = timezone.localdate()
    live = Extension.objects.filter(archived_at__isnull=True)
    total = live.count()

    coordinators = User.objects.filter(role=Role.COORDINATOR, extension_id=OuterRef("pk")).filter(signin_filter())
    with_coordinator = live.filter(Exists(coordinators)).count()

    recent_sheets = _scope_attendance(user, None, today - timedelta(days=SILENT_DAYS), today).filter(
        extension_id=OuterRef("pk")
    )
    silent = live.filter(~Exists(recent_sheets)).order_by("name")

    dates = _gathering_dates(user, None, start, end)
    attendance_rows = (
        _scope_attendance(user).filter(attendance_date__in=dates)
        .values("extension__name").annotate(present=Coalesce(Sum("total_present"), 0))
        .order_by("-present", "extension__name")[:EXTENSION_BAR_LIMIT]
    )
    people_rows = (
        live.annotate(n=Count("members", filter=~Q(members__status=Status.ARCHIVED.value)))
        .order_by("-n", "name")[:EXTENSION_BAR_LIMIT]
    )
    people_per_extension = [{"name": title_case(ext.name), "count": ext.n} for ext in people_rows]
    return {
        "total": total,
        "with_coordinator": with_coordinator,
        "without_coordinator": total - with_coordinator,
        "silent": [{"id": ext.pk, "name": title_case(ext.name)} for ext in silent],
        "silent_days": SILENT_DAYS,
        "attendance_per_extension": [
            {"name": title_case(row["extension__name"]), "count": row["present"]} for row in attendance_rows
        ],
        "people_per_extension": people_per_extension,
        "people_more": max(0, total - len(people_per_extension)),
    }


def giving_stats(user, extension_id=None, start=None, end=None):
    """Return the giving numbers for a leader (None for a member). Amounts are Decimal, never float.

    Only finalized reports on uploaded sheets, FROZEN ``tithes_total`` and ``offering_amount``, reports on the
    same date added together. Keys: ``reports`` (count), ``tithes`` and ``offering`` (totals), ``total``
    (both), their ``*_display`` text with ``CURRENCY``, and ``trend`` (one row per gathering date, oldest
    first: ``date``, ``label``, ``tithes``, ``offering``, ``total``). ``labels`` and ``totals`` feed a sparkline.
    Amounts are shown on leader dashboards only and are never written to ``audit_log`` or ``notification.data``.
    """
    if user.role not in LEADER_ROLES:
        return None
    zero = Decimal("0")
    dates = _gathering_dates(user, extension_id, start, end)
    reports = _scope_tithes(user, extension_id).filter(attendance__attendance_date__in=dates)
    rows = reports.values("attendance__attendance_date").annotate(
        tithes=Sum("tithes_total"), offering=Sum("offering_amount"), n=Count("pk")
    ).order_by()
    by_date = {row["attendance__attendance_date"]: row for row in rows}

    trend = []
    for day in dates:
        row = by_date.get(day, {})
        tithes, offering = row.get("tithes") or zero, row.get("offering") or zero
        trend.append({"date": day, "label": _day_label(day), "tithes": tithes, "offering": offering, "total": tithes + offering})
    tithes_sum = sum((t["tithes"] for t in trend), zero)
    offering_sum = sum((t["offering"] for t in trend), zero)
    return {
        "reports": sum(row["n"] for row in by_date.values()),
        "tithes": tithes_sum,
        "offering": offering_sum,
        "total": tithes_sum + offering_sum,
        "tithes_display": _peso(tithes_sum),
        "offering_display": _peso(offering_sum),
        "total_display": _peso(tithes_sum + offering_sum),
        "trend": trend,
        "labels": [t["label"] for t in trend],
        "totals": [t["total"] for t in trend],
    }


def prayer_summary(user, extension_id=None):
    """Return the prayer card numbers for the people who handle prayer (None for everyone else).

    Scoping is ONLY ``PrayerRequest.objects.inbox_for(user)`` (archived never listed). ``open`` counts requests
    not yet completed (open and replied), ``completed`` the rest, and ``latest`` lists the newest
    ``PRAYER_LATEST`` unfinished ones as ``{"id", "label", "status", "status_label", "created_at", "extension"}``.
    ``label`` comes from ``sender_label`` (anonymous stays anonymous); the request text and ``user_id`` are never
    returned. When an admin or super-admin filters to an extension whose coordinator can sign in, the inbox
    holds none of its requests (that coordinator handles them), so ``scope_note`` explains it instead.
    """
    if not can_use_inbox(user):
        return None
    inbox = PrayerRequest.objects.inbox_for(user)
    scope_note = ""
    if extension_id and user.role != Role.COORDINATOR:
        inbox = inbox.filter(extension_id=extension_id)
        has_coordinator = User.objects.filter(role=Role.COORDINATOR, extension_id=extension_id).filter(signin_filter())
        if has_coordinator.exists():
            scope_note = PRAYER_SCOPE_NOTE

    unfinished = inbox.exclude(status=PrayerRequest.Status.COMPLETED)
    newest = unfinished.select_related("user", "extension").order_by("-created_at", "-pk")[:PRAYER_LATEST]
    return {
        "open": unfinished.count(),
        "completed": inbox.filter(status=PrayerRequest.Status.COMPLETED).count(),
        "latest": [
            {
                "id": req.pk,
                "label": sender_label(req, user),
                "status": req.status,
                "status_label": req.get_status_display(),
                "created_at": req.created_at,
                "extension": title_case(req.extension.name),
            }
            for req in newest
        ],
        "scope_note": scope_note,
    }


def upcoming_events(user, limit=5):
    """Return the next events the person may see, soonest first, as plain dicts.

    Scoping is ONLY ``events.permissions.visible_events``. Upcoming means ``Coalesce(end_at, start_at) >= now``
    (an event still running counts); cancelled events are left out. Each row: ``id``, ``title``, ``start_at``,
    ``end_at``, ``is_global`` and ``extension`` (the extension name, "" for a global event).
    """
    now = timezone.now()
    soonest = (
        visible_events(user).annotate(ends_at=Coalesce("end_at", "start_at")).filter(ends_at__gte=now)
        .select_related("extension").order_by("start_at", "pk")
    )
    # `is_cancelled` is a Python property on Event, so cancelled rows are dropped here; the bounded fetch
    # keeps a long list of cancelled events from being loaded.
    events = [event for event in soonest[: limit * 4] if not event.is_cancelled][:limit]
    return [
        {
            "id": event.pk,
            "title": event.title,
            "start_at": event.start_at,
            "end_at": event.end_at,
            "is_global": event.extension_id is None,
            "extension": title_case(event.extension.name) if event.extension_id else "",
        }
        for event in events
    ]

def dashboard_tiles(attendance, people, extensions, period_start=None):
    """Return the four headline tiles of a leader dashboard as ``{"label", "value", "hint"}`` dicts.

    People, the last gathering, average attendance, and either how many extensions reported lately (admin and
    super-admin: ``extensions`` is set) or the first-timers total (a coordinator has no extension comparison).
    Every number is also text, so no tile depends on color.
    """
    if not attendance or not people:
        return []
    count = attendance["count"]
    latest = attendance["latest"]
    joined = people["new_in_period"]
    gatherings = f"{count} gathering{'' if count == 1 else 's'}"

    tiles = [
        {
            "label": "People",
            "value": people["total"],
            "hint": f"{joined} new since {_day_label(period_start)}" if joined is not None and period_start else f"{people['active']} active",
        },
        {
            "label": "Last gathering",
            "value": latest["present"] if latest else "–",
            "hint": latest["label"] if latest else "No uploaded sheets yet",
        },
        {
            "label": "Average attendance",
            "value": f"{attendance['average']:g}" if count else "–",
            "hint": f"Over {gatherings}" if count else "No uploaded sheets yet",
        },
    ]
    if extensions:
        reporting = extensions["total"] - len(extensions["silent"])
        tiles.append({
            "label": "Extensions reporting",
            "value": f"{reporting} of {extensions['total']}",
            "hint": f"Sent a sheet in the last {extensions['silent_days']} days",
        })
    else:
        tiles.append({
            "label": "First-timers",
            "value": attendance["first_timers"],
            "hint": f"Over {gatherings}" if count else "No uploaded sheets yet",
        })
    return tiles


def _ellipsize(text, limit=LABEL_LIMIT):
    """Return ``text`` cut to ``limit`` characters with an ellipsis (SVG text cannot truncate itself)."""
    return text if len(text) <= limit else text[: limit - 1].rstrip() + "…"


def _donut_chart(title, caption, rows):
    """Return donut geometry plus its title, caption and a one-line text description (for ``aria-label``)."""
    donut = charts.donut([(row["label"], row["count"]) for row in rows], radius=60, stroke=22)
    summary = ", ".join(f"{item['label']} {item['value']}" for item in donut["items"])
    return {**donut, "title": title, "caption": caption, "aria": f"{title}: {summary}" if summary else f"{title}: no data"}


def _hbar_chart(title, caption, rows, more=0):
    """Return horizontal-bar geometry for ``{"name", "count"}`` rows, with long names shortened.

    Each drawn row keeps the full name in ``full`` (the tooltip) and the number in ``value`` (the text).
    """
    chart = charts.hbar([(_ellipsize(row["name"]), row["count"]) for row in rows], max_width=HBAR_WIDTH, row_height=36, gap=10)
    for drawn, source in zip(chart["rows"], rows):
        drawn["full"] = source["name"]
    summary = ", ".join(f"{row['name']} {row['count']}" for row in rows)
    return {**chart, "title": title, "caption": caption, "more": more, "aria": f"{title}: {summary}" if summary else f"{title}: no data"}


def leader_charts(attendance, people, extensions, giving, period_given=False):
    """Return every leader chart as a prepared dict for the ``_chart_*`` partials (None for a member).

    Keys: ``attendance_line``, ``attendance_columns`` (men and women by age network: the newest gathering, or
    every gathering when a period was asked for), ``donut_roles``, ``donut_status``, ``giving_spark`` and, for
    admin and super-admin only, ``people_bars`` and ``attendance_bars`` (None for a coordinator). Each dict is the
    geometry from ``charts.py`` plus ``title``, ``caption`` and ``aria`` (the whole chart as one sentence); the
    drawn values are always in the dict too, so the partials can print them as text.
    """
    if attendance is None or people is None:
        return None
    points, count = attendance["points"], attendance["count"]

    line = charts.line_chart(attendance["values"], attendance["labels"], CHART_WIDTH, CHART_HEIGHT)
    line.update(
        title="Attendance",
        caption="Total present per gathering in the period" if period_given else f"Total present, last {count} gathering{'' if count == 1 else 's'}",
        aria="Attendance: " + (", ".join(f"{p['label']} {p['present']}" for p in points) if points else "no uploaded sheets yet"),
    )

    network = attendance["network"] if period_given else attendance["network_latest"]
    series = [row["label"] for row in network["rows"]]
    groups = [(col["label"], [row["values"][i] for row in network["rows"]]) for i, col in enumerate(network["columns"])]
    columns = charts.column_groups(groups, CHART_WIDTH, CHART_HEIGHT)
    latest = attendance["latest"]
    for group in columns["groups"]:
        for bar in group["bars"]:
            bar["series_label"] = series[bar["series"]] if bar["series"] < len(series) else ""   # for the tooltip
            bar["cx"] = round(bar["x"] + bar["width"] / 2, 2)                                      # centre, for the value text
    parts = [
        f"{label}: " + ", ".join(f"{name} {bar['value']:g}" for name, bar in zip(series, group["bars"]))
        for (label, _values), group in zip(groups, columns["groups"])
    ]
    columns.update(
        title="Men and women by age group",
        caption="Every gathering in the period" if period_given else (f"Latest gathering, {latest['label']}" if latest else "No uploaded sheets yet"),
        series=series,
        aria="Men and women by age group: " + ("no data" if columns["empty"] else "; ".join(parts)),
    )

    spark = charts.sparkline(giving["totals"], 120, 32) if giving else None
    if spark is not None:
        spark["items"] = [{"label": row["label"], "display": _peso(row["total"])} for row in giving["trend"]]
        spark["aria"] = "Giving per gathering: " + (", ".join(f"{i['label']} {i['display']}" for i in spark["items"]) or "no reports yet")

    return {
        "attendance_line": line,
        "attendance_columns": columns,
        "donut_roles": _donut_chart("People by role", "Not archived", people["roles"]),
        "donut_status": _donut_chart("People by status", "Everyone, including archived", people["statuses"]),
        "giving_spark": spark,
        "people_bars": _hbar_chart(
            "People per extension", "Biggest, not archived. Not changed by the picker.", extensions["people_per_extension"], extensions["people_more"]
        ) if extensions else None,
        "attendance_bars": _hbar_chart(
            "Attendance per extension", "Total present in the period. Not changed by the picker.", extensions["attendance_per_extension"]
        ) if extensions else None,
    }


# ----------------------------------------------------------------- helpers
def _bars(rows, display=str, short=None):
    """Turn (label, value) rows into ``Bar`` objects scaled against the biggest value."""
    top = max((float(value) for _label, value in rows), default=0.0)
    make_short = short or display
    bars = []
    for label, value in rows:
        pct = round(float(value) / top * 100) if top else 0
        if value and pct < MIN_BAR_PERCENT:
            pct = MIN_BAR_PERCENT
        bars.append(Bar(label=label, display=display(value), short=make_short(value), pct=pct))
    return bars


def _day_label(day):
    """Return a short date such as ``Oct 12``."""
    return f"{day:%b} {day.day}"


def _peso(value):
    """Return an amount as text, for example ``₱1,500.00``."""
    return f"{CURRENCY}{value:,.2f}"


def _compact(value):
    """Return a short number for narrow charts: 950, 1.5k, 2.3m."""
    number = float(value)
    for limit, suffix in ((1_000_000, "m"), (1_000, "k")):
        if number >= limit:
            return f"{number / limit:.1f}".rstrip("0").rstrip(".") + suffix
    return f"{number:.0f}"

===== apps/dashboards/charts.py =====
"""dashboards.charts: pure chart geometry. No Django, no database, no HTML.

Every function takes plain numbers and returns plain dicts and lists that a template (or the A4
print page) draws as inline SVG. Nothing here knows about colors or CSS classes: the screen
templates add token classes (``stroke-accent``...), the print templates add grayscale attributes.

Rules shared by all functions:

* Empty input, a single value and all zeros never divide by zero. They return a dict with
  ``"empty": True`` (or a sensible flat shape) so a template can draw "No data yet" instead.
* Values may be ``int``, ``float`` or ``Decimal``; they are turned into ``float`` ONLY to work out
  positions. Money is never stored or returned as a float: the caller keeps its Decimal and
  prints that as text.
* Negative, ``None`` and non-finite values are treated as 0 (these charts show counts and totals).
* Coordinates are rounded to 2 decimals so the SVG stays small and tests are stable.
* SVG y grows downward, so a bigger value has a SMALLER y.
"""
import math

# Space kept free around the plot area for tick labels (left, bottom) and the stroke (top, right).
PAD_TOP = 16
PAD_RIGHT = 16
PAD_BOTTOM = 32
PAD_LEFT = 40

#: Ticks aim for about this many intervals on the y axis.
TARGET_TICKS = 4
#: A non-zero bar is never thinner than this share of the full width (a tiny value must still show).
MIN_BAR_PERCENT = 4


# ------------------------------------------------------------------ helpers
def _num(value):
    """Return ``value`` as a safe float: None, NaN, infinity and negatives become 0.0."""
    try:
        number = float(value)
    except (TypeError, ValueError):
        return 0.0
    if not math.isfinite(number) or number < 0:
        return 0.0
    return number


def _r(number):
    """Round a coordinate to 2 decimals (and avoid ``-0.0``)."""
    return round(number, 2) + 0.0


def _labels(labels, count):
    """Return exactly ``count`` text labels: missing ones become 1, 2, 3... so a mark always has a name."""
    given = [str(label) for label in (labels or [])][:count]
    return given + [str(i + 1) for i in range(len(given), count)]


def _tick_text(number):
    """Return a tick as text: ``40`` not ``40.0``, ``0.5`` stays ``0.5``."""
    if float(number).is_integer():
        return str(int(number))
    return f"{number:g}"


def _nice_step(raw_step, integer_only):
    """Round a raw step up to 1, 2 or 5 times a power of ten (never below 1 when ``integer_only``)."""
    if raw_step <= 0:
        return 1.0
    magnitude = 10 ** math.floor(math.log10(raw_step))
    for factor in (1, 2, 5, 10):
        step = factor * magnitude
        if step >= raw_step:
            break
    return max(step, 1.0) if integer_only else step


def y_scale(top_value, integer_only=True):
    """Return ``(ticks, ceiling)``: round tick values from 0 and the top tick every chart scales to.

    ``top_value`` is the biggest value drawn. Zero (or less) still gives ``[0, 1]``, so an empty
    chart has an axis and nothing divides by zero.
    """
    top = _num(top_value)
    if top <= 0:
        return [0.0, 1.0], 1.0
    step = _nice_step(top / TARGET_TICKS, integer_only)
    count = max(1, math.ceil(top / step - 1e-9))
    ticks = [round(step * i, 10) for i in range(count + 1)]
    return ticks, ticks[-1]


def _y_ticks(ticks, ceiling, top, plot_height):
    """Turn tick values into ``{"value", "label", "y"}`` rows for the grid lines."""
    return [
        {"value": tick, "label": _tick_text(tick), "y": _r(top + plot_height - tick / ceiling * plot_height)}
        for tick in ticks
    ]


def _plot_box(width, height):
    """Return the plot rectangle inside the padding as a dict (left, top, right, bottom, width, height)."""
    left, top = PAD_LEFT, PAD_TOP
    right, bottom = max(left + 1, width - PAD_RIGHT), max(top + 1, height - PAD_BOTTOM)
    return {"left": left, "top": top, "right": right, "bottom": bottom, "width": right - left, "height": bottom - top}


# ---------------------------------------------------------------- line chart
def line_chart(values, labels, width=640, height=240):
    """Geometry for a line chart (attendance over the last gatherings).

    Returns ``{"width", "height", "plot", "points", "polyline", "area", "y_ticks", "baseline_y",
    "ceiling", "empty"}``:

    * ``points``: one ``{"x", "y", "label", "value"}`` per value, oldest first. ``label`` and ``value``
      are what the template writes in the ``<title>`` tooltip and in the text list.
    * ``polyline``: the ``points`` attribute string (``"x,y x,y"``); ``area``: a closed path
      (``d`` attribute) under the line, or ``""`` with fewer than two points.
    * ``y_ticks``: grid lines, 0 at the bottom up to ``ceiling``.
    * With no values ``points`` is ``[]`` and ``empty`` is True; one value sits in the middle.
    """
    numbers = [_num(v) for v in (values or [])]
    box = _plot_box(width, height)
    ticks, ceiling = y_scale(max(numbers, default=0))
    names = _labels(labels, len(numbers))
    count = len(numbers)

    points = []
    for index, number in enumerate(numbers):
        x = box["left"] + (box["width"] / 2 if count == 1 else index * box["width"] / (count - 1))
        y = box["bottom"] - number / ceiling * box["height"]
        points.append({"x": _r(x), "y": _r(y), "label": names[index], "value": values[index] if values else 0})

    polyline = " ".join(f"{p['x']},{p['y']}" for p in points)
    area = ""
    if count >= 2:
        area = (
            "M" + " L".join(f"{p['x']},{p['y']}" for p in points)
            + f" L{points[-1]['x']},{_r(box['bottom'])} L{points[0]['x']},{_r(box['bottom'])} Z"
        )
    return {
        "width": width, "height": height, "plot": box,
        "points": points, "polyline": polyline, "area": area,
        "y_ticks": _y_ticks(ticks, ceiling, box["top"], box["height"]),
        "baseline_y": _r(box["bottom"]), "ceiling": ceiling, "empty": count == 0,
    }


# ------------------------------------------------------------ grouped columns
def column_groups(groups, width=640, height=240, gap_ratio=0.3):
    """Geometry for grouped columns (for example age network groups, each with men and women).

    ``groups`` is a list of ``(label, [value per series])`` pairs (a ``{"label", "values"}`` dict also
    works). Every group should have the same number of series; a short one is padded with zeros.

    Returns ``{"width", "height", "plot", "groups", "y_ticks", "baseline_y", "ceiling", "series_count",
    "empty"}``. Each group is ``{"label", "label_x", "bars"}`` and each bar is
    ``{"x", "y", "width", "height", "value", "series", "group"}`` (``series`` is the index inside the
    group, so the template can color series 0 and 1 differently and write the legend).
    A zero value still returns a bar with ``height`` 0, so its tooltip and text stay in step.
    ``empty`` is True when there are no groups or every value is 0.
    """
    rows = []
    for group in groups or []:
        label, values = (group["label"], group["values"]) if isinstance(group, dict) else group
        rows.append((str(label), [_num(v) for v in values]))

    box = _plot_box(width, height)
    series_count = max((len(values) for _label, values in rows), default=0)
    top = max((v for _label, values in rows for v in values), default=0)
    ticks, ceiling = y_scale(top)

    group_count = len(rows)
    slot = box["width"] / group_count if group_count else box["width"]
    inner = slot * (1 - gap_ratio)                       # the part of a slot the bars use
    bar_width = inner / series_count if series_count else 0

    drawn = []
    for g_index, (label, values) in enumerate(rows):
        start = box["left"] + g_index * slot + (slot - inner) / 2
        bars = []
        for s_index in range(series_count):
            value = values[s_index] if s_index < len(values) else 0.0
            bar_height = value / ceiling * box["height"]
            bars.append({
                "x": _r(start + s_index * bar_width), "y": _r(box["bottom"] - bar_height),
                "width": _r(bar_width), "height": _r(bar_height),
                "value": value, "series": s_index, "group": label,
            })
        drawn.append({"label": label, "label_x": _r(box["left"] + g_index * slot + slot / 2), "bars": bars})

    return {
        "width": width, "height": height, "plot": box, "groups": drawn,
        "y_ticks": _y_ticks(ticks, ceiling, box["top"], box["height"]),
        "baseline_y": _r(box["bottom"]), "ceiling": ceiling, "series_count": series_count,
        "empty": top <= 0,
    }


# --------------------------------------------------------------------- donut
def donut(segments, radius=60, stroke=18):
    """Geometry for a donut (or a progress ring with two segments).

    ``segments`` is a list of ``(label, value)`` pairs (a ``{"label", "value"}`` dict also works).
    Each arc is a ``<circle>`` with the same ``r``: draw it with ``stroke-dasharray`` and
    ``stroke-dashoffset`` from here, inside a group rotated -90 degrees so the first arc starts at 12 o'clock.

    Returns ``{"size", "cx", "cy", "r", "stroke", "circumference", "total", "items", "arcs", "empty"}``.
    ``items`` has every segment (``label``, ``value``, ``pct`` with one decimal, ``index``) in the order given,
    for the text list and legend. ``arcs`` has the same fields plus ``dasharray`` and ``dashoffset`` for the
    segments above zero only. With a total of 0 there are no arcs and ``empty`` is True (draw the track alone).
    """
    parsed = []
    for segment in segments or []:
        label, value = (segment["label"], segment["value"]) if isinstance(segment, dict) else segment
        parsed.append((str(label), value, _num(value)))

    circumference = 2 * math.pi * radius
    total = sum(number for _l, _v, number in parsed)
    size = 2 * radius + stroke
    items, arcs, used = [], [], 0.0
    for index, (label, value, number) in enumerate(parsed):
        pct = round(number / total * 100, 1) if total else 0.0
        item = {"label": label, "value": value, "pct": pct, "index": index}
        items.append(item)
        if number <= 0 or not total:
            continue
        length = number / total * circumference
        arcs.append({
            **item,
            "dasharray": f"{_r(length)} {_r(max(circumference - length, 0))}",
            "dashoffset": _r(-used),
        })
        used += length
    return {
        "size": size, "cx": _r(size / 2), "cy": _r(size / 2), "r": radius, "stroke": stroke,
        "circumference": _r(circumference), "total": total, "items": items, "arcs": arcs, "empty": total <= 0,
    }


# ------------------------------------------------------- horizontal bars
def hbar(items, max_width=300, row_height=22, gap=8):
    """Geometry for horizontal bars (people or attendance per extension).

    ``items`` is a list of ``(label, value)`` pairs, already sorted by the caller. The biggest value gets
    ``max_width``; a non-zero value is never thinner than ``MIN_BAR_PERCENT`` of it.

    Returns ``{"rows", "max", "max_width", "height", "empty"}``; each row is ``{"label", "value", "width",
    "pct", "y", "height"}`` (``y`` stacks rows top to bottom; ``height`` of the whole chart is ``height``).
    All zeros give rows with ``width`` 0 and ``empty`` True.
    """
    parsed = [(str(label), value, _num(value)) for label, value in (items or [])]
    top = max((number for _l, _v, number in parsed), default=0.0)
    rows = []
    for index, (label, value, number) in enumerate(parsed):
        pct = number / top * 100 if top else 0.0
        if number > 0:
            pct = max(pct, MIN_BAR_PERCENT)
        rows.append({
            "label": label, "value": value, "pct": round(pct, 1), "width": _r(pct / 100 * max_width),
            "y": index * (row_height + gap), "height": row_height,
        })
    total_height = len(rows) * (row_height + gap) - gap if rows else 0
    return {"rows": rows, "max": top, "max_width": max_width, "height": max(total_height, 0), "empty": top <= 0}


# ----------------------------------------------------------------- sparkline
def sparkline(values, width=120, height=32):
    """Geometry for a tiny line with no axes (the giving trend).

    Returns ``{"width", "height", "points", "polyline", "last", "min", "max", "empty"}``. ``last`` is the
    final point (for a dot) or None. A flat series (all equal, or all zeros) is drawn straight through the
    middle; one value is a single point in the middle.
    """
    numbers = [_num(v) for v in (values or [])]
    pad = 3                                              # keeps the stroke and the last dot inside the box
    inner_w, inner_h = max(width - 2 * pad, 1), max(height - 2 * pad, 1)
    low, high = (min(numbers), max(numbers)) if numbers else (0.0, 0.0)
    span = high - low
    count = len(numbers)

    points = []
    for index, number in enumerate(numbers):
        x = pad + (inner_w / 2 if count == 1 else index * inner_w / (count - 1))
        y = pad + inner_h / 2 if span == 0 else pad + inner_h - (number - low) / span * inner_h
        points.append({"x": _r(x), "y": _r(y), "value": values[index]})
    return {
        "width": width, "height": height, "points": points,
        "polyline": " ".join(f"{p['x']},{p['y']}" for p in points),
        "last": points[-1] if points else None, "min": low, "max": high, "empty": count == 0,
    }

The extension filter (``?extension=<id>``) is read here for the super-admin and the admin ONLY. The raw text goes to
the services, which validate it (a bad value means "All extensions"). The coordinator and member views never read it,
so a forged ``?extension=`` cannot move a coordinator out of their own extension.

from django.shortcuts import render
from django.views.decorators.cache import never_cache

from apps.accounts.decorators import role_required
from apps.accounts.models import Role

from .services import VERSION_COOKIE, get_dashboard_context


def _render(request, template, use_filter=False):
    """Render a role's dashboard template with the shared dashboard context.

    ``use_filter`` is True only for the super-admin and admin views: then ``?extension=`` is passed on as raw text.
    """
    extension = request.GET.get("extension") if use_filter else None
    dashboard = get_dashboard_context(
        request.user, request.COOKIES.get(VERSION_COOKIE), request.path, extension=extension
    )
    return render(request, template, {"dashboard": dashboard})

from datetime import date, timedelta

from apps.accounts.models import Extension, PersonType, Status, User
from apps.attendance.age import AgeNetwork
from apps.attendance.models import Attendance
from apps.tithes.models import TithesOffering

GENDERS = ("male", "female")
GROUPS = ("first_timers", "guests", "members")


def make_person(role, extension=None, status=Status.ACTIVE, **extra):
    """Create an activated person who can sign in (`must_change_password=False`, `member_account`)."""
    return User.objects.create_user(
        password="pass-12345", first_name="test", last_name=role.replace("_", ""), role=role,
        extension=extension, status=status, must_change_password=False,
        person_type=PersonType.MEMBER_ACCOUNT, **extra,
    )


def make_extension(name):
    """Create an extension by name."""
    return Extension.objects.create(name=name)


def summary_json(cells=None):
    """Build a frozen-summary dict; ``cells`` maps (group, gender, network) to a count, the rest are 0."""
    summary = {
        group: {gender: {network: 0 for network in AgeNetwork.values} for gender in GENDERS} for group in GROUPS
    }
    for (group, gender, network), number in (cells or {}).items():
        summary[group][gender][network] = number
    return summary


def add_uploaded_sheet(extension, day, present, first_timers=0, cells=None, label="Sunday Service", summary=True):
    """Create an UPLOADED sheet with frozen totals and (unless ``summary=False``) a frozen breakdown."""
    return Attendance.objects.create(
        extension=extension, attendance_date=day, label=label, status=Attendance.Status.UPLOADED,
        total_present=present, first_timers_count=first_timers, members_present=present - first_timers,
        summary=summary_json(cells) if summary else None, uploaded_at=timezone.now(),
    )


def add_draft_sheet(extension, day, label="Sunday Service"):
    """Create a DRAFT sheet (no frozen numbers); dashboards must ignore it."""
    return Attendance.objects.create(extension=extension, attendance_date=day, label=label)


def add_finalized_report(sheet, tithes, offering):
    """Create a FINALIZED tithes report on ``sheet`` with the given frozen amounts."""
    tithes, offering = Decimal(tithes), Decimal(offering)
    return TithesOffering.objects.create(
        attendance=sheet, status=TithesOffering.Status.FINALIZED, offering_amount=offering,
        tithes_total=tithes, total_amount=tithes + offering, breakdown_total=tithes + offering,
        difference=Decimal("0"), tithers_count=1, finalized_at=timezone.now(),
    )


def sundays(count, last=date(2026, 10, 18)):
    """Return ``count`` weekly dates ending on ``last``, oldest first."""
    return [last - timedelta(weeks=count - 1 - i) for i in range(count)]

class ResolveExtensionTests(TestCase):

        self.alpha, self.beta = make_extension("alpha"), make_extension("beta")

def _resolve_name(name, role):
    """Return the URL for a url name (or ROLE_HOME), or None when the route does not exist (yet).

    ``name`` may list alternatives separated by ``|`` ("app:a|app:b"): the first one that
    resolves wins. Use it while a route's final name is not settled.
    """
    if "|" in name:
        for alternative in name.split("|"):
            href = _resolve_name(alternative.strip(), role)
            if href:
                return href
        return None
    if name == ROLE_HOME:
        # Imported here to avoid a circular import with accounts.decorators.
        from apps.accounts.decorators import ROLE_HOME_NAMES

        name = ROLE_HOME_NAMES.get(role)
        if not name:
            return None
    try:
        return reverse(name)
    except NoReverseMatch:
        return None


def _resolve_href(item, role):
    """Return the item's URL, or None when its route does not exist (yet)."""
    return _resolve_name(item.url_name, role)


def _is_active(href, path, exact):
    """True when the current path belongs to this link."""
    if not href:
        return False
    if path == href:
        return True
    return not exact and href != "/" and path.startswith(href)


def _match_length(owned, path):
    """Length of the longest owned prefix that the path falls under (0 = no match)."""
    return max((len(h) for h in owned if h and h != "/" and path.startswith(h)), default=0)


def build_tabs(section, role, path):
    """Return the visible tabs of one hub as plain dicts (label, href, disabled, active).

    Tabs the role may not see are left out; a tab whose route does not exist yet
    is kept but ``disabled`` so the hub shape is visible while views are built.
    Only the tab with the most specific match for ``path`` is active.
    """
    tabs, scores = [], []
    for tab in SECTION_TABS.get(section, []):
        if role not in tab.roles:
            continue
        href = _resolve_name(tab.url_name, role)
        owned = [href] + [_resolve_name(name, role) for name in tab.also] + list(tab.prefixes)
        scores.append(_match_length(owned, path))
        tabs.append({"label": tab.label, "href": href or "#", "disabled": href is None, "active": False})
    best = max(scores, default=0)
    for tab, score in zip(tabs, scores):
        tab["active"] = best > 0 and score == best
    return tabs


def hub_tabs_for(role, path):
    """Return the tabs of whichever hub the current page belongs to ([] when it is in none)."""
    for section in SECTION_TABS:
        tabs = build_tabs(section, role, path)
        if any(tab["active"] for tab in tabs):
            return tabs
    return []


def _entry(item, role, path):
    """Turn a NavItem into the plain dict the templates render."""
    if item.tabs:
        # A hub links to its first usable tab and is lit up on any of its tabs.
        tabs = build_tabs(item.tabs, role, path)
        first = next((tab for tab in tabs if not tab["disabled"]), None)
        href = first["href"] if first else None
        active = any(tab["active"] for tab in tabs)
    else:
        href = _resolve_href(item, role)
        active = _is_active(href, path, item.exact)
    return {
        "label": item.label,
        "icon": item.icon,
        "href": href or "#",
        "disabled": href is None,
        "active": active,
    }


def build_nav(role, path):
    """Build what the templates need for one user on one page.

    Returns a dict with:
      sections     -> [{"heading": str, "items": [entry, ...]}, ...] (sidebar, drawer)
      bottom_items -> the entries shown on the mobile bottom bar
      more_items   -> every other entry, shown in the bottom bar's "More" sheet
    """
    visible = [item for item in NAV_ITEMS if role in item.roles]
    entries = [_entry(item, role, path) for item in visible]

    sections = []
    for item, entry in zip(visible, entries):
        if not sections or sections[-1]["heading"] != item.group:
            sections.append({"heading": item.group, "items": []})
        sections[-1]["items"].append(entry)

    flagged = [(item, entry) for item, entry in zip(visible, entries) if item.bottom]
    on_bar = flagged[:BOTTOM_BAR_SLOTS]
    # Compare before relabelling, so the More sheet still lists everything that is not on the bar.
    more_items = [entry for entry in entries if entry not in [e for _, e in on_bar]]
    # The bar gets the short label (when set); the sidebar and More sheet keep the full one.
    bottom_items = [{**entry, "label": item.short or entry["label"]} for item, entry in on_bar]

    return {"sections": sections, "bottom_items": bottom_items, "more_items": more_items}

# URL names, resolved at request time. These are the real dashboard routes

# dict for the "Home" link, so the sidebar and the redirects always agree.
ROLE_HOME_NAMES = {
    Role.SUPER_ADMIN: "dashboards:superadmin",
    Role.ADMIN: "dashboards:admin",
    Role.COORDINATOR: "dashboards:coordinator",
    Role.MEMBER: "dashboards:member",

def home_url_for(user):
    """Return the landing URL for this user's role."""
    name = ROLE_HOME_NAMES.get(user.role)
    return reverse(name) if name else settings.LOGIN_REDIRECT_URL


def role_required(*roles):
    """Allow only these roles; anyone else goes to their own home page."""
    def decorator(view):
        @login_required
        @wraps(view)
        def wrapper(request, *args, **kwargs):
            if request.user.role not in roles:
                return redirect(home_url_for(request.user))
            return view(request, *args, **kwargs)
        return wrapper
    return decorator

MAX_END_NOTE = 500           # characters in a final note
END_REQUEST_DAYS_KEY = "buddy_end_request_days"   # site_setting key
DEFAULT_END_REQUEST_DAYS = 7                      # used when the setting is missing or invalid

NOT_FOUND = "That request was not found."

PARTNER_GOALS_LIVE = "live"            # a live partner: the goals are shown
PARTNER_GOALS_INACTIVE = "inactive"    # partner cannot sign in: goals hidden
PARTNER_GOALS_NONE = "no_partner"      # no un-ended partnership at all


class CandidateState(models.TextChoices):
    """What the Request button should do for one person in the candidate list."""

    AVAILABLE = "available", "Available"
    PENDING_SENT = "pending_sent", "Request sent"
    PENDING_RECEIVED = "pending_received", "Asked you"
    HAS_BUDDY = "has_buddy", "Already has a buddy"


# ------------------------------------------------------------------ helpers
def _lock_users(user_ids):
    """Lock these people in ascending id order and return `{id: User}` (missing ids are absent).

    The rows come back freshly read, so permission checks on them are checks on the locked row.
    No `select_related`: `extension` is a nullable foreign key.
    """
    ordered = sorted(set(user_ids))
    rows = User.objects.select_for_update().filter(pk__in=ordered).order_by("pk")
    return {user.pk: user for user in rows}


def partner_of(row, user):
    """The other person in a partnership row. Raises ValueError when `user` is not in it."""
    if row.user_a_id == user.pk:
        return row.user_b
    if row.user_b_id == user.pk:
        return row.user_a
    raise ValueError("That person is not in this partnership.")


def is_live(row):
    """True when both people in the row can still sign in. A row that is not live is 'inactive'."""
    return row.user_a.can_sign_in and row.user_b.can_sign_in


def _active_rows(user, *, lock=False):
    """Un-ended partnership rows that include `user`, newest first."""
    rows = UserBuddy.objects.filter(ended_at__isnull=True).filter(Q(user_a=user) | Q(user_b=user))
    if lock:
        rows = rows.select_for_update()
    return rows.order_by("-pk")


def active_partnership(user):
    """The person's newest un-ended partnership row (live or inactive), or None."""
    return _active_rows(user).first()


def live_partnership(user):
    """The person's un-ended partnership whose two people can both sign in, or None."""
    for row in _active_rows(user):
        if is_live(row):
            return row
    return None


def end_dead_partnerships(user, *, actor=None):
    """End every un-ended partnership of `user` that is not live. Returns how many were ended.

    `ended_by` stays NULL (nobody chose this) and no final note is stored. Call it inside an
    atomic block that already holds the lock on `user`.
    """
    ended = 0
    with transaction.atomic():
        for row in _active_rows(user, lock=True):
            if is_live(row):
                continue
            row.ended_at = timezone.now()
            row.save(update_fields=["ended_at"])
            log_action(
                "buddy.end", actor=actor or user, entity_type="user_buddy", entity_id=row.pk,
                after={"reason": "partner_cannot_sign_in"},
            )
            ended += 1
    return ended


# ----------------------------------------------------------------- requests
def send_request(actor, to_user_id):
    """Ask another person to be your buddy. Returns the new pending `BuddyRequest`.

    Refused when it is yourself, when either person cannot sign in or already has a LIVE buddy,
    when you already have 3 pending requests, or when a request between the two of you is
    already pending (in either direction). An inactive partnership of yours is ended first.
    """
    if actor.pk == to_user_id:
        raise ServiceError("You cannot send a buddy request to yourself.")
    with transaction.atomic():
        locked = _lock_users([actor.pk, to_user_id])
        me, target = locked.get(actor.pk), locked.get(to_user_id)
        if me is None or not can_use_buddy(me):
            raise ServiceError("You cannot use Buddy right now.")
        if target is None or not can_use_buddy(target):
            raise ServiceError("That person is not available.")

        end_dead_partnerships(me)
        if live_partnership(me):
            raise ServiceError("You already have a buddy.")
        if live_partnership(target):
            raise ServiceError("That person already has a buddy.")

        pending = BuddyRequest.objects.filter(status=BuddyRequest.Status.PENDING)
        between = pending.filter(
            Q(from_user=me, to_user=target) | Q(from_user=target, to_user=me)
        ).first()
        if between is not None:
            if between.from_user_id == me.pk:
                raise ServiceError("You already sent that person a request.")
            raise ServiceError("That person already asked you. Accept it under Requests for you.")
        if pending.filter(from_user=me).count() >= MAX_PENDING_REQUESTS:
            raise ServiceError(f"You can have up to {MAX_PENDING_REQUESTS} pending requests. Cancel one first.")

        try:
            with transaction.atomic():
                request = BuddyRequest.objects.create(from_user=me, to_user=target)
        except IntegrityError:
            # The unique constraint is the last safety net for a duplicate.
            raise ServiceError("There is already a pending request between you two.")

        log_action(
            "buddy.request", actor=me, entity_type="buddy_request", entity_id=request.pk,
            after={"to_user_id": target.pk},
        )
        notify(
            target, BUDDY_REQUEST, actor=me, entity_type="buddy_request", entity_id=request.pk,
            data={"request_id": request.pk}, dedupe_key=f"buddy_request:{request.pk}",
        )
    return request


def accept_request(actor, request_id):
    """Accept a request sent to you. Returns the new (or already existing) `UserBuddy` row.

    Locks both people, re-checks that neither has a LIVE buddy, creates the partnership with
    the two ids sorted, and cancels every other pending request that involves either person.
    Accepting twice is quiet: the second call returns the same partnership.
    """
    with transaction.atomic():
        found = BuddyRequest.objects.filter(pk=request_id).first()
        if found is None or found.to_user_id != actor.pk:
            raise ServiceError(NOT_FOUND)
        locked = _lock_users([found.from_user_id, found.to_user_id])
        # People are locked first, then the request (same order everywhere).
        request = BuddyRequest.objects.select_for_update().filter(pk=request_id).first()
        me, sender = locked.get(found.to_user_id), locked.get(found.from_user_id)
        first_id, second_id = sorted([found.from_user_id, found.to_user_id])

        if request is None or request.to_user_id != actor.pk:
            raise ServiceError(NOT_FOUND)
        if request.status == BuddyRequest.Status.ACCEPTED:
            existing = UserBuddy.objects.filter(user_a_id=first_id, user_b_id=second_id, ended_at__isnull=True).first()
            if existing is not None:
                return existing
            raise ServiceError("This request was already accepted.")
        if request.status != BuddyRequest.Status.PENDING:
            raise ServiceError("This request is no longer open.")
        if me is None or not can_use_buddy(me):
            raise ServiceError("You cannot use Buddy right now.")
        if sender is None or not can_use_buddy(sender):
            raise ServiceError("That person can no longer take part in Buddy.")

        # An inactive partnership never blocks a new one: it is ended here.
        end_dead_partnerships(me)
        end_dead_partnerships(sender, actor=me)
        if live_partnership(me):
            raise ServiceError("You already have a buddy.")
        if live_partnership(sender):
            raise ServiceError("That person already has a buddy.")

        now = timezone.now()
        row = UserBuddy.objects.create(user_a_id=first_id, user_b_id=second_id)
        request.status = BuddyRequest.Status.ACCEPTED
        request.responded_at = now
        request.save(update_fields=["status", "responded_at"])
        cancelled = (
            BuddyRequest.objects.filter(status=BuddyRequest.Status.PENDING)
            .filter(Q(from_user_id__in=[first_id, second_id]) | Q(to_user_id__in=[first_id, second_id]))
            .update(status=BuddyRequest.Status.CANCELLED, responded_at=now)
        )

        log_action(
            "buddy.accept", actor=me, entity_type="user_buddy", entity_id=row.pk,
            after={"request_id": request.pk, "user_a_id": first_id, "user_b_id": second_id,
                   "cancelled_requests": cancelled},
        )
        notify(
            sender, BUDDY_ACCEPTED, actor=me, entity_type="user_buddy", entity_id=row.pk,
            data={"buddy_id": row.pk}, dedupe_key=f"buddy_accepted:{row.pk}",
        )
    return row


def _close_request(actor, request_id, *, side, new_status, action):
    """Shared body of decline (recipient) and cancel (sender). Repeating it is quiet."""
    with transaction.atomic():
        me = _lock_users([actor.pk]).get(actor.pk)
        if me is None or not can_use_buddy(me):
            raise ServiceError("You cannot use Buddy right now.")
        request = BuddyRequest.objects.select_for_update().filter(pk=request_id).first()
        if request is None or getattr(request, f"{side}_id") != me.pk:
            raise ServiceError(NOT_FOUND)
        if request.status == new_status:
            return request
        if request.status != BuddyRequest.Status.PENDING:
            raise ServiceError("This request is no longer open.")
        request.status = new_status
        request.responded_at = timezone.now()
        request.save(update_fields=["status", "responded_at"])
        log_action(action, actor=me, entity_type="buddy_request", entity_id=request.pk)
    return request


def decline_request(actor, request_id):
    """The recipient turns a request down. There is no cooldown: the sender may ask again."""
    return _close_request(
        actor, request_id, side="to_user", new_status=BuddyRequest.Status.DECLINED, action="buddy.decline",
    )


def cancel_request(actor, request_id):
    """The sender takes back a request that is still pending."""
    return _close_request(
        actor, request_id, side="from_user", new_status=BuddyRequest.Status.CANCELLED, action="buddy.cancel",
    )


def incoming_requests(user):
    """Pending requests sent to `user` by people who can still sign in, newest first."""
    return list(
        BuddyRequest.objects.filter(to_user=user, status=BuddyRequest.Status.PENDING)
        .filter(signable_q("from_user__"))
        .select_related("from_user", "from_user__extension")
        .order_by("-created_at", "-pk")
    )


def outgoing_requests(user):
    """Pending requests `user` sent, newest first. All of them show, so each can be cancelled."""
    return list(
        BuddyRequest.objects.filter(from_user=user, status=BuddyRequest.Status.PENDING)
        .select_related("to_user", "to_user__extension")
        .order_by("-created_at", "-pk")
    )


# --------------------------------------------------------------- candidates
def _page_number(value):
    """A page number from the query string: a whole number from 1 up, anything else is 1."""
    try:
        return max(1, min(int(value), MAX_PAGE))
    except (TypeError, ValueError):
        return 1


def candidates(actor, query="", page=1):
    """People `actor` could ask to be a buddy, one page of at most 20.

    DELIBERATE EXCEPTION to `User.objects.for_user()`: that stops at a coordinator's own
    extension, but a buddy may come from any extension, so this looks at every person who can
    sign in. To keep that safe it needs `can_use_buddy`, never returns the actor, caps a page at
    20 people, and gives back only id, display name, extension name and a state. No contact
    details, role or account id leave this function.

    Empty search (or fewer than 3 characters): the actor's own extension; admins and
    super-admins have no extension, so they see everyone. 3 or more characters: every
    extension, every word typed must match the first, middle or last name.

    Returns `{"rows", "page", "has_next", "has_previous", "query", "scope", "too_short"}`
    where `scope` is "search", "everyone" or "extension".
    """
    if not can_use_buddy(actor):
        raise ServiceError("You cannot use Buddy right now.")
    words = (query or "").split()
    text = " ".join(words)[:100]
    words = text.split()
    searching = len(text) >= SEARCH_MIN_LENGTH

    people = User.objects.filter(signable_q()).exclude(pk=actor.pk).select_related("extension")
    if searching:
        scope = "search"
        for word in words:
            word = clean_text(word)
            people = people.filter(
                Q(first_name__icontains=word) | Q(middle_name__icontains=word) | Q(last_name__icontains=word)
            )
    elif actor.role in (Role.ADMIN, Role.SUPER_ADMIN):
        scope = "everyone"
    else:
        scope = "extension"
        people = people.filter(extension_id=actor.extension_id) if actor.extension_id else people.none()

    page = _page_number(page)
    offset = (page - 1) * PAGE_SIZE
    chunk = list(people.order_by("last_name", "first_name", "pk")[offset:offset + PAGE_SIZE + 1])
    has_next = len(chunk) > PAGE_SIZE
    chunk = chunk[:PAGE_SIZE]

    states = _states_for(actor, [person.pk for person in chunk])
    rows = [
        {
            "id": person.pk,
            "name": person.full_name,
            "extension": person.extension.name if person.extension_id else "",
            "state": states.get(person.pk, CandidateState.AVAILABLE),
        }
        for person in chunk
    ]
    return {
        "rows": rows, "page": page, "has_next": has_next, "has_previous": page > 1,
        "query": text, "scope": scope, "too_short": bool(text) and not searching,
    }


def _states_for(actor, person_ids):
    """`{person id: CandidateState}` for the people on one page, in three queries."""
    if not person_ids:
        return {}
    busy = set()
    rows = (
        UserBuddy.objects.filter(ended_at__isnull=True)
        .filter(Q(user_a_id__in=person_ids) | Q(user_b_id__in=person_ids))
        .select_related("user_a", "user_b")
    )
    for row in rows:
        if is_live(row):                      # an inactive partnership does not make anyone busy
            busy.update((row.user_a_id, row.user_b_id))
    pending = BuddyRequest.objects.filter(status=BuddyRequest.Status.PENDING)
    sent = set(pending.filter(from_user=actor, to_user_id__in=person_ids).values_list("to_user_id", flat=True))
    received = set(pending.filter(to_user=actor, from_user_id__in=person_ids).values_list("from_user_id", flat=True))

    states = {}
    for pid in person_ids:
        if pid in busy:
            states[pid] = CandidateState.HAS_BUDDY
        elif pid in sent:
            states[pid] = CandidateState.PENDING_SENT
        elif pid in received:
            states[pid] = CandidateState.PENDING_RECEIVED
        else:
            states[pid] = CandidateState.AVAILABLE
    return states


# ------------------------------------------------------------- partner card
# Where each number may live in whatever `faith.streaks.streak_snapshot` returns. The snapshot's
# exact shape was not visible when this brick was written, so the lookup accepts the model field
# names, short names and a nested form. A test reads a real snapshot, so a mismatch shows up there.
_STREAK_PATHS = {
    "open_current": (("current_open_streak",), ("open_current",), ("current_open",), ("open", "current")),
    "open_best": (("longest_open_streak",), ("open_longest",), ("longest_open",), ("open_best",), ("open", "longest"), ("open", "best")),
    "read_current": (("current_read_streak",), ("read_current",), ("current_read",), ("read", "current")),
    "read_best": (("longest_read_streak",), ("read_longest",), ("longest_read",), ("read_best",), ("read", "longest"), ("read", "best")),
}


def _lookup(source, path):
    """Follow `path` through mappings or attributes; None when any step is missing."""
    for key in path:
        source = source.get(key) if isinstance(source, Mapping) else getattr(source, key, None)
        if source is None:
            return None
    return source


def _partner_streaks(partner):
    """The partner's four streak numbers, or None when the faith streak engine is not there."""
    try:
        from apps.faith.streaks import streak_snapshot
    except ImportError:          # Brick 1 not installed yet: the card simply shows dashes
        return None
    snapshot = streak_snapshot(partner, timezone.localdate())
    result = {}
    for name, paths in _STREAK_PATHS.items():
        value = next((v for v in (_lookup(snapshot, p) for p in paths) if v is not None), 0)
        result[name] = value
    return result


def partner_card(user):
    """What the partner card shows for `user`, or None when they have no LIVE partner.

    Read-only: name, extension, since when, and the partner's open and Bible streaks. Goals,
    notes and messages are never part of it.
    """
    row = live_partnership(user)
    if row is None:
        return None
    partner = partner_of(row, user)
    return {
        "partnership_id": row.pk,
        "partner_id": partner.pk,
        "name": partner.full_name,
        "extension": partner.extension.name if partner.extension_id else "",
        "since": row.started_at,
        "streaks": _partner_streaks(partner),
    }


# ------------------------------------------------------------ ending (Brick 7)
def _clean_end_note(note):
    """A final note trimmed; None when empty. Over 500 characters is refused, never cut."""
    text = (note or "").strip()
    if len(text) > MAX_END_NOTE:
        raise ServiceError(f"A final note can have up to {MAX_END_NOTE} characters.")
    return text or None


def _find_row(user, *, live=None):
    """The person's newest un-ended row that is live (True), inactive (False) or either (None)."""
    for row in _active_rows(user):
        if live is None or is_live(row) is live:
            return row
    return None


def _lock_row(actor, *, live=None, missing="You do not have a buddy right now."):
    """Lock both people (ascending id), then the actor's un-ended row. Returns (row, me, partner).

    Everything is re-checked on the locked rows: the actor may still use Buddy, the row is still
    un-ended, and it is still live / inactive as the caller expects (`live=None` accepts both).
    Call it inside `transaction.atomic()`.
    """
    found = _find_row(actor, live=live)
    if found is None:
        raise ServiceError(missing)
    locked = _lock_users([found.user_a_id, found.user_b_id])
    me = locked.get(actor.pk)
    partner = locked.get(found.user_b_id if found.user_a_id == actor.pk else found.user_a_id)
    if me is None or not can_use_buddy(me):
        raise ServiceError("You cannot use Buddy right now.")
    row = UserBuddy.objects.select_for_update().filter(pk=found.pk, ended_at__isnull=True).first()
    if row is None:
        raise ServiceError("This partnership has already ended.")
    if partner is None:
        raise ServiceError(missing)
    if live is not None and partner.can_sign_in is not live:
        raise ServiceError("Your buddy's account just changed. Reload the page and try again.")
    return row, me, partner


def request_end(actor, note=None):
    """Ask your buddy to end the partnership, with an optional final note (up to 500 characters).

    Live partnership only. A second request changes nothing (the first one, by either person,
    stays), so a double tap is quiet. The partner gets a `buddy_end_requested` bell item.
    """
    text = _clean_end_note(note)
    with transaction.atomic():
        row, me, partner = _lock_row(actor, live=True)
        if row.end_requested_at is not None:
            return row
        now = timezone.now()
        row.end_requested_by = me
        row.end_requested_at = now
        row.end_request_note = text
        row.save(update_fields=["end_requested_by", "end_requested_at", "end_request_note"])
        log_action(
            "buddy.end_request", actor=me, entity_type="user_buddy", entity_id=row.pk,
            after={"requested_by_id": me.pk},
        )
        # The time is in the key because the dedupe key is permanent: a request that was cancelled
        # and made again must still reach the partner.
        notify(
            partner, BUDDY_END_REQUESTED, actor=me, entity_type="user_buddy", entity_id=row.pk,
            data={"buddy_id": row.pk}, dedupe_key=f"buddy_end_requested:{row.pk}:{now:%Y%m%d%H%M%S%f}",
        )
    return row


def cancel_end_request(actor):
    """Take back your own end request. Nothing pending: quiet. Someone else's request: refused."""
    with transaction.atomic():
        row, me, _partner = _lock_row(actor)
        if row.end_requested_at is None:
            return row
        if row.end_requested_by_id != me.pk:
            raise ServiceError("Only the person who asked can cancel the request.")
        row.end_requested_by = None
        row.end_requested_at = None
        row.end_request_note = None
        row.save(update_fields=["end_requested_by", "end_requested_at", "end_request_note"])
        log_action("buddy.end_cancel", actor=me, entity_type="user_buddy", entity_id=row.pk)
    return row


def agree_end(actor, note=None):
    """The OTHER person agrees to end it, with their own optional note. Ends the partnership.

    Stores `ended_by` = the requester, `final_note` = the requester's note and
    `partner_final_note` = this person's note. The requester gets a `buddy_ended` bell item.
    """
    text = _clean_end_note(note)
    with transaction.atomic():
        row, me, partner = _lock_row(actor, live=True)
        if row.end_requested_at is None:
            raise ServiceError("Your buddy has not asked to end the partnership.")
        if row.end_requested_by_id == me.pk:
            raise ServiceError("You asked to end this partnership. Your buddy has to agree.")
        row.ended_at = timezone.now()
        row.ended_by = partner                      # only two people: the requester is the partner
        row.final_note = row.end_request_note
        row.partner_final_note = text
        row.save(update_fields=["ended_at", "ended_by", "final_note", "partner_final_note"])
        log_action(
            "buddy.end", actor=me, entity_type="user_buddy", entity_id=row.pk,
            after={"mode": "agreed", "ended_by_id": partner.pk},
        )
        notify(
            partner, BUDDY_ENDED, actor=me, entity_type="user_buddy", entity_id=row.pk,
            data={"buddy_id": row.pk}, dedupe_key=f"buddy_ended:{row.pk}",
        )
    return row


def end_without_approval(actor, note=None):
    """End a partnership whose other person can no longer sign in. No agreement is needed.

    Refused while the partner can still sign in. Stores `ended_by` = actor and `final_note`.
    Nobody is notified: the partner cannot sign in to read it.
    """
    text = _clean_end_note(note)
    with transaction.atomic():
        if _find_row(actor, live=False) is None and live_partnership(actor) is not None:
            raise ServiceError("Your buddy can still sign in, so ending needs their agreement.")
        row, me, _partner = _lock_row(actor, live=False, missing="You have no partnership to end.")
        row.ended_at = timezone.now()
        row.ended_by = me
        row.final_note = text
        row.save(update_fields=["ended_at", "ended_by", "final_note"])
        log_action(
            "buddy.end", actor=me, entity_type="user_buddy", entity_id=row.pk,
            after={"mode": "inactive_partner", "ended_by_id": me.pk},
        )
    return row


def partnership_state(user):
    """What the End controls on the Buddy tab need, or None when `user` has no un-ended partnership.

    `mode` is "live" (nothing pending), "waiting" (you asked), "asked" (your buddy asked, so
    `request_note` is theirs for you to read) or "inactive" (your buddy can no longer sign in).
    """
    row = live_partnership(user) or active_partnership(user)
    if row is None:
        return None
    partner = partner_of(row, user)
    if not is_live(row):
        mode = "inactive"
    elif row.end_requested_at is None:
        mode = "live"
    elif row.end_requested_by_id == user.pk:
        mode = "waiting"
    else:
        mode = "asked"
    return {
        "mode": mode,
        "partner_name": partner.full_name,
        "requested_at": row.end_requested_at,
        "request_note": row.end_request_note if mode == "asked" else None,
        "auto_close_days": end_request_days(),
    }


def end_request_days():
    """Days an end request may wait before it closes by itself (site setting, default 7).

    A missing, non-numeric or out-of-range (1 to 365) value falls back to the default.
    """
    # (verify) If `core` already has a `site_setting(key, default)` helper, call it here instead.
    from apps.core.models import SiteSetting

    row = SiteSetting.objects.filter(key=END_REQUEST_DAYS_KEY).first()
    raw = getattr(row, "value", None)
    if isinstance(raw, dict):
        raw = raw.get("value")
    try:
        days = int(raw)
    except (TypeError, ValueError):
        return DEFAULT_END_REQUEST_DAYS
    return days if 1 <= days <= 365 else DEFAULT_END_REQUEST_DAYS


def stale_end_request_ids(now=None, days=None):
    """Ids of un-ended partnerships whose end request is older than `days` (the job and its dry run)."""
    now = now or timezone.now()
    days = end_request_days() if days is None else days
    return list(
        UserBuddy.objects.filter(
            ended_at__isnull=True, end_requested_at__isnull=False,
            end_requested_at__lt=now - timedelta(days=days),
        ).order_by("pk").values_list("pk", flat=True)
    )


def close_stale_end_requests(now=None, days=None):
    """End every partnership whose end request went unanswered for too long. Returns the count.

    The requester's wish is honoured: `ended_by` = the requester, `final_note` = their note, no
    `partner_final_note`. Each row is re-checked under its lock, so the job is safe to run twice
    and safe next to a person who agrees at the same moment. Logged as a system job.
    """
    now = now or timezone.now()
    days = end_request_days() if days is None else days
    closed = 0
    for pk in stale_end_request_ids(now, days):
        with transaction.atomic():
            found = UserBuddy.objects.filter(pk=pk).first()
            if found is None:
                continue
            _lock_users([found.user_a_id, found.user_b_id])
            row = UserBuddy.objects.select_for_update().filter(
                pk=pk, ended_at__isnull=True, end_requested_at__isnull=False,
                end_requested_at__lt=now - timedelta(days=days),
            ).first()
            if row is None:
                continue
            row.ended_at = now
            row.ended_by_id = row.end_requested_by_id
            row.final_note = row.end_request_note
            row.save(update_fields=["ended_at", "ended_by", "final_note"])
            log_action(
                "buddy.end", entity_type="user_buddy", entity_id=row.pk,
                after={"mode": "end_request_expired", "ended_by_id": row.end_requested_by_id},
                source=AuditLog.Source.SYSTEM_JOB,
            )
            closed += 1
    return closed

def partner_goals_for(viewer):
    """What the "Their goals" screen shows `viewer`: their LIVE partner's goals, nothing else.

===== apps/faith/overview.py =====
"""faith.overview: the numbers behind the goal pictures (Brick 4).

Two entry points, both read-only and both taking ``today`` as a parameter (never ``now()``):

* ``goal_overview(goal, today)``: one goal's streaks, completion, 84-day heatmap, last 8 weeks, best
  weekday and a month grid;
* ``global_overview(user, today)``: every goal that is not archived (completed ones stay), as a
  heatmap, "perfect days", week and month rates, and the "opened the app" and "read the Bible" rows.

Each entry point is a thin database wrapper around a pure ``build_*`` function that takes plain
Python values (goals, sets of done dates). The pure functions never touch the database, so the
maths can be tested without any rows.

Rules shared by every number here:

* A day that is not scheduled never counts for or against anyone.
* TODAY IS STILL OPEN until midnight: a scheduled goal that is not ticked today is left out of the
  "expected" counts and never breaks a streak. Once ticked it counts as done. This is the same rule
  the Today screen uses for a goal's own streak.
* A streak is a run of scheduled days that were done, so a weekly goal's streak counts weeks' worth
  of occurrences, not calendar days.
* Check-in notes are never read. Only dates and the ``done`` flag leave the database.

Privacy: nothing in this module writes to ``audit_log`` or ``notification.data``. Goal titles are
returned to the caller (the page the owner is looking at) and nowhere else.
"""
import calendar
from collections import defaultdict
from datetime import timedelta
from fractions import Fraction

from apps.faith import schedule
from apps.faith.models import Goal, GoalCheckin, UserDailyActivity
from apps.faith.streaks import streak_snapshot

HEATMAP_WEEKS = 12          # 12 weeks x 7 days = the 84-day heatmap (Monday first, ends on this week's Sunday)
BAR_WEEKS = 8               # weeks shown in the per-goal bar chart
ACTIVITY_DAYS = 28          # days shown in the "opened the app" / "read the Bible" strips
MIN_BAR_PERCENT = 4         # a non-zero bar is never invisible (same rule as the dashboard charts)
MIN_WEEKDAY_SAMPLE = 2      # a weekday needs this many settled days before it can be called "best"
RING_CIRCUMFERENCE = 251.33  # 2 * pi * 40, for the r=40 progress ring drawn by faith/_ring.html

STATUS_ACTIVE = "active"
STATUS_ARCHIVED = "archived"

WEEKDAY_NAMES = ("Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday")
WEEKDAY_SHORT = ("Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun")

# Words used in the aria-label / title of a per-goal cell.
GOAL_STATE_TEXT = {
    "done": "done",
    "missed": "missed",
    "off": "not scheduled",
    "open": "still open today",
    "future": "upcoming",
}


# ------------------------------------------------------------------ small helpers
def _week_start(day):
    """Return the Monday of the week that holds ``day``."""
    return day - timedelta(days=day.weekday())


def _day_label(day):
    """Return a day as text for a label, for example ``Mon 3 Mar``."""
    return f"{day:%a} {day.day} {day:%b}"


def _short_date(day):
    """Return a short date such as ``Oct 12`` (same style as the dashboard charts)."""
    return f"{day:%b} {day.day}"


def _rate(done, expected):
    """Return a whole percent (half rounds up), or None when nothing was expected."""
    if not expected:
        return None
    return (200 * done + expected) // (2 * expected)


def _runs(hits):
    """Return ``(current, best)`` run lengths for booleans listed oldest first.

    ``current`` is the run that ends at the last item; ``best`` is the longest run anywhere.
    """
    run = best = 0
    for hit in hits:
        run = run + 1 if hit else 0
        best = max(best, run)
    return run, best


def _cell(day, state, text, **extra):
    """Return one picture cell: the date, a state name and the text for its title and aria-label."""
    return {"date": day, "state": state, "label": f"{_day_label(day)}: {text}", **extra}


def heatmap_days(today):
    """Return the 84 dates of the heatmap, Monday of the first week first, in week-by-week order.

    The last week is the current one, so days after ``today`` are in the list (shown as upcoming).
    """
    first = _week_start(today) - timedelta(weeks=HEATMAP_WEEKS - 1)
    return [first + timedelta(days=i) for i in range(HEATMAP_WEEKS * 7)]


def make_ring(pct):
    """Return what the progress ring template needs: percent, stroke dash length and the middle text.

    ``pct`` None means "nothing to measure yet": an empty ring with a dash in the middle.
    """
    shown = "-" if pct is None else f"{pct}%"
    pct = max(0, min(100, pct or 0))
    return {
        "pct": pct,
        "dash": f"{RING_CIRCUMFERENCE * pct / 100:.2f}",
        "circumference": f"{RING_CIRCUMFERENCE:.2f}",
        "text": shown,
    }


def time_label(value):
    """Return a clock time as ``6:30 AM``."""
    return f"{(value.hour % 12) or 12}:{value.minute:02d} {'AM' if value.hour < 12 else 'PM'}"


def when_summary(goal):
    """Return the goal's time as text: ``All day``, ``6:00 AM`` or ``6:00 AM - 7:00 AM``."""
    if goal.start_time is None:
        return "All day"
    if goal.end_time is None:
        return time_label(goal.start_time)
    return f"{time_label(goal.start_time)} - {time_label(goal.end_time)}"


def repeat_summary(goal):
    """Return how the goal repeats as text, for example ``Every day`` or ``Weekly: Mon, Wed``."""
    if goal.repeat_type == "weekly":
        return "Weekly: " + ", ".join(WEEKDAY_SHORT[d] for d in sorted(goal.weekdays or []))
    if goal.end_date is not None and goal.end_date == goal.start_date:
        return "One time"
    return "Every day"


# ------------------------------------------------------------------ one goal
def _settled(goal, done_dates, today):
    """Return ``[(day, hit)]``, oldest first, for the goal's scheduled days up to today.

    Today is included only once it has been ticked (an unticked today is still open).
    """
    end = min(goal.end_date, today) if goal.end_date else today
    if end < goal.start_date:
        return []
    rows = []
    for day in schedule.scheduled_dates(goal, goal.start_date, end):
        hit = day in done_dates
        if hit or day != today:
            rows.append((day, hit))
    return rows


def _goal_cell(goal, day, done_dates, today, number=False):
    """Return the cell for one day of one goal: done, missed, off, open (today, not ticked) or future."""
    if day > today:
        state = "future"
    elif not schedule.is_scheduled(goal, day):
        state = "off"
    elif day in done_dates:
        state = "done"
    else:
        state = "open" if day == today else "missed"
    extra = {"number": day.day} if number else {}
    return _cell(day, state, GOAL_STATE_TEXT[state], **extra)


def _week_bars(settled, today):
    """Return the last ``BAR_WEEKS`` weeks as column-chart rows (label, display, short, pct), oldest first.

    The height is the share done that week; the number above the column is ``done/expected``.
    """
    totals = defaultdict(lambda: [0, 0])
    for day, hit in settled:
        week = totals[_week_start(day)]
        week[0] += hit
        week[1] += 1
    bars = []
    for back in range(BAR_WEEKS - 1, -1, -1):
        start = _week_start(today) - timedelta(weeks=back)
        done, expected = totals.get(start, (0, 0))
        pct = _rate(done, expected) or 0
        if done and pct < MIN_BAR_PERCENT:
            pct = MIN_BAR_PERCENT
        bars.append({
            "label": _short_date(start),
            "display": f"{done} of {expected} done" if expected else "nothing scheduled",
            "short": f"{done}/{expected}" if expected else "-",
            "pct": pct,
        })
    return bars


def _best_weekday(settled):
    """Return the weekday with the best done-rate, or None when there is not enough to say.

    A weekday needs ``MIN_WEEKDAY_SAMPLE`` settled days, and at least one of them done.
    A tie goes to the earlier weekday.
    """
    totals = defaultdict(lambda: [0, 0])
    for day, hit in settled:
        entry = totals[day.weekday()]
        entry[0] += hit
        entry[1] += 1
    best = None
    for weekday, (done, expected) in sorted(totals.items()):
        if expected < MIN_WEEKDAY_SAMPLE or not done:
            continue
        if best is None or Fraction(done, expected) > best[0]:
            best = (Fraction(done, expected), weekday, done, expected)
    if best is None:
        return None
    _share, weekday, done, expected = best
    return {"name": WEEKDAY_NAMES[weekday], "rate": _rate(done, expected), "done": done, "expected": expected}


def _month_grid(goal, done_dates, today):
    """Return this month as weeks of seven cells (Monday first). Days of other months are ``pad`` cells."""
    weeks = []
    for week in calendar.Calendar(firstweekday=0).monthdatescalendar(today.year, today.month):
        row = []
        for day in week:
            if day.month != today.month:
                row.append({"date": day, "state": "pad", "label": "", "number": ""})
            else:
                row.append(_goal_cell(goal, day, done_dates, today, number=True))
        weeks.append(row)
    return {"label": f"{calendar.month_name[today.month]} {today.year}", "weeks": weeks}


def build_goal_overview(goal, done_dates, today):
    """Return everything the goal detail page draws, from plain values (no database).

    ``done_dates`` is the set of dates on which the goal was ticked (``done=True`` rows only).
    """
    settled = _settled(goal, done_dates, today)
    done = sum(1 for _day, hit in settled if hit)
    expected = len(settled)
    current, best = _runs(hit for _day, hit in settled)
    rate = _rate(done, expected)
    return {
        "streak": {"current": current, "best": best},
        "done": done,
        "expected": expected,
        "rate": rate,
        "ring": make_ring(rate),
        "has_checkins": bool(done_dates),
        "heatmap": [_goal_cell(goal, day, done_dates, today) for day in heatmap_days(today)],
        "weeks": _week_bars(settled, today),
        "best_weekday": _best_weekday(settled),
        "month": _month_grid(goal, done_dates, today),
    }


def goal_overview(goal, today):
    """Return the detail-page numbers for one goal (one query: the goal's done dates).

    The caller must already have checked that the viewer owns the goal.
    """
    done_dates = set(
        GoalCheckin.objects.filter(goal=goal, done=True).values_list("checkin_date", flat=True)
    )
    return build_goal_overview(goal, done_dates, today)


# ------------------------------------------------------------------ all goals together
def _global_cell(day, sched, done, today):
    """Return the cell for one day across all goals.

    States: ``off`` (nothing scheduled), ``future``, ``missed`` (nothing done, day closed), ``open``
    (nothing done yet today), ``partial`` (``level`` 1-3 by the share done) and ``perfect`` (all done).
    """
    if day > today:
        return _cell(day, "future", "upcoming")
    scheduled = sched.get(day, 0)
    if not scheduled:
        return _cell(day, "off", "no goals scheduled")
    ticked = done.get(day, 0)
    text = f"{ticked} of {scheduled} done"
    if ticked == scheduled:
        return _cell(day, "perfect", text)
    if not ticked:
        return _cell(day, "open" if day == today else "missed", text)
    if 3 * ticked <= scheduled:
        level = 1
    elif 3 * ticked <= 2 * scheduled:
        level = 2
    else:
        level = 3
    return _cell(day, "partial", text, level=level)


def _window_rate(sched, done, today, days):
    """Return ``{"done", "expected", "rate"}`` for the last ``days`` days ending today (today still open rule)."""
    open_today = sched.get(today, 0) - done.get(today, 0)
    ticked = expected = 0
    for back in range(days):
        day = today - timedelta(days=back)
        ticked += done.get(day, 0)
        expected += sched.get(day, 0) - (open_today if day == today else 0)
    return {"done": ticked, "expected": expected, "rate": _rate(ticked, expected)}


def _perfect_days(sched, done, today):
    """Return ``{"current", "best", "total"}`` for perfect days (every scheduled goal done).

    Only days with at least one scheduled goal take part; a day with nothing scheduled neither
    adds to nor breaks a run. Today is skipped until every goal scheduled today is done.
    """
    hits = []
    for day in sorted(sched):
        perfect = done.get(day, 0) == sched[day]
        if day == today and not perfect:
            continue
        hits.append(perfect)
    current, best = _runs(hits)
    return {"current": current, "best": best, "total": sum(hits)}


def build_global(goals, done_by_goal, today):
    """Return the history page numbers for these goals, from plain values (no database).

    ``goals`` are the goals to include (archived ones already left out) and ``done_by_goal`` maps a goal
    id to the set of dates it was ticked.
    """
    sched = defaultdict(int)
    done = defaultdict(int)
    for goal in goals:
        end = min(goal.end_date, today) if goal.end_date else today
        if end < goal.start_date:
            continue
        hits = done_by_goal.get(goal.id, ())
        for day in schedule.scheduled_dates(goal, goal.start_date, end):
            sched[day] += 1
            if day in hits:
                done[day] += 1
    return {
        "has_goals": bool(goals),
        "goal_count": len(goals),
        "heatmap": [_global_cell(day, sched, done, today) for day in heatmap_days(today)],
        "perfect": _perfect_days(sched, done, today),
        "week": _window_rate(sched, done, today, 7),
        "month": _window_rate(sched, done, today, 30),
    }


def _snapshot_numbers(snapshot):
    """Return ``{"open": {"current", "best"}, "read": {"current", "best"}}`` from ``streak_snapshot``.

    VERIFY: Brick 1 only promised "current and longest for both streaks". This is the one place that
    knows the real shape, so adapt the candidate names below if the lookup raises.
    """
    def pick(*names):
        for name in names:
            if isinstance(snapshot, dict) and name in snapshot:
                return snapshot[name]
            if hasattr(snapshot, name):
                return getattr(snapshot, name)
        raise AttributeError(f"streak_snapshot has none of {names}")

    return {
        "open": {
            "current": pick("current_open_streak", "open_current", "current_open"),
            "best": pick("longest_open_streak", "open_longest", "longest_open"),
        },
        "read": {
            "current": pick("current_read_streak", "read_current", "current_read"),
            "best": pick("longest_read_streak", "read_longest", "longest_read"),
        },
    }


def streak_cards(user, today):
    """Return the open and Bible-reading streaks (current and best) for a streak widget."""
    return _snapshot_numbers(streak_snapshot(user, today))


def build_activity(rows, streaks, today):
    """Return the two activity strips ("Opened the app", "Read the Bible") for the last ``ACTIVITY_DAYS`` days.

    ``rows`` are ``(activity_date, opened_app, chapters_read)`` tuples. A cell is ``done``, ``missed`` or,
    for today when nothing has happened yet, ``open``.
    """
    opened = {day for day, was_opened, _chapters in rows if was_opened}
    read = {day for day, _opened, chapters in rows if chapters}
    days = [today - timedelta(days=back) for back in range(ACTIVITY_DAYS - 1, -1, -1)]
    specs = (
        ("open", "Opened the app", "calendar-days", opened, "opened the app", "did not open the app"),
        ("read", "Read the Bible", "book-open", read, "read the Bible", "did not read"),
    )
    result = []
    for key, label, icon, hit_days, yes, no in specs:
        cells = []
        for day in days:
            if day in hit_days:
                cells.append(_cell(day, "done", yes))
            elif day == today:
                cells.append(_cell(day, "open", "not yet today"))
            else:
                cells.append(_cell(day, "missed", no))
        result.append({
            "key": key,
            "label": label,
            "icon": icon,
            "cells": cells,
            "days": sum(1 for day in days if day in hit_days),
            "window": ACTIVITY_DAYS,
            "current": streaks[key]["current"],
            "best": streaks[key]["best"],
        })
    return result


def global_overview(user, today):
    """Return the history page numbers for one person: goals plus the two activity rows.

    Archived goals are left out; completed goals are included. Two goal queries at most (the goals, then
    every done date of those goals in one query; the perfect-day streak needs the whole history).
    """
    goals = list(Goal.objects.filter(user=user).exclude(status=STATUS_ARCHIVED))
    done_by_goal = defaultdict(set)
    if goals:
        rows = GoalCheckin.objects.filter(goal_id__in=[g.id for g in goals], done=True)
        for goal_id, day in rows.values_list("goal_id", "checkin_date"):
            done_by_goal[goal_id].add(day)
    result = build_global(goals, done_by_goal, today)
    first = today - timedelta(days=ACTIVITY_DAYS - 1)
    activity_rows = list(
        UserDailyActivity.objects.filter(user=user, activity_date__range=(first, today))
        .values_list("activity_date", "opened_app", "chapters_read")
    )
    result["activity"] = build_activity(activity_rows, streak_cards(user, today), today)
    return result


# ------------------------------------------------------------------ dashboard widget
def build_today(goals_today, done_ids):
    """Return the "goals today" widget numbers from goals scheduled today and the ids already ticked.

    ``next_up`` lists up to three goals still to do: all-day goals first, then timed ones in time order.
    """
    total = len(goals_today)
    ticked = sum(1 for goal in goals_today if goal.id in done_ids)
    pending = sorted(
        (g for g in goals_today if g.id not in done_ids),
        key=lambda g: (g.start_time is not None, g.start_time or 0, g.title.lower()),
    )
    pct = _rate(ticked, total)
    return {
        "total": total,
        "done": ticked,
        "remaining": total - ticked,
        "all_done": bool(total) and ticked == total,
        "ring": make_ring(pct),
        "next_up": [
            {"id": g.id, "title": g.title, "when": when_summary(g) if g.start_time else ""}
            for g in pending[:3]
        ],
        "more": max(0, len(pending) - 3),
    }


def today_summary(user, today):
    """Return the "goals today" widget numbers for a person (two small queries)."""
    goals = list(Goal.objects.filter(user=user, status=STATUS_ACTIVE))
    scheduled = [g for g in goals if schedule.is_scheduled(g, today)]
    done_ids = set()
    if scheduled:
        done_ids = set(
            GoalCheckin.objects.filter(goal_id__in=[g.id for g in scheduled], checkin_date=today, done=True)
            .values_list("goal_id", flat=True)
        )
    return build_today(scheduled, done_ids)

===== apps/events/permissions.py =====
"""events.permissions: ONE place that decides who may see and do what with an event.

Views start from `visible_events(actor)` (and answer 404 for anything outside it), and every
service calls `require(...)` again: hiding a button is not access control.

Who may do what (A = admin or super-admin, C = coordinator, M = member):
  see a global event             everyone who can sign in
  see an extension event         A any | C and M of THAT extension only
  create / edit / cancel global  A only
  create / edit / cancel ext.    A any | C own extension only | M never

Events belong to the EXTENSION, never to a coordinator account, so these rules compare
`actor.extension_id` with `event.extension_id` and the history survives a reassignment.

These functions answer ROLE and SCOPE only, plus the state rule that a cancelled event
cannot be edited again (`can_edit`, `can_cancel`). Timing rules (a started event cannot be
moved, an ended event cannot be cancelled) live in `services.py` as friendly ServiceErrors.
"""
from django.core.exceptions import PermissionDenied
from django.db.models import Q

from apps.accounts.models import Extension, Role

from .models import Event

VIEW = "view"
EDIT = "edit"
CANCEL = "cancel"

_ADMINS = (Role.SUPER_ADMIN, Role.ADMIN)


def _signed_in(actor):
    """True for a real person who can sign in (not anonymous, not archived, not suspended)."""
    return actor is not None and getattr(actor, "is_authenticated", False) and actor.is_active


def _own_extension(actor, extension_id):
    """True when `actor` belongs to the extension with this id."""
    return extension_id is not None and actor.extension_id == extension_id


# ----------------------------------------------------------------- seeing
def visible_events(actor):
    """The events `actor` may see, as a queryset (the ONE place that narrows events by who asks).

    Global events for everyone; extension events only for that extension's people; admins see
    all. Archived events are left out. Anyone else gets an empty queryset, so a view that does
    `get_object_or_404(visible_events(request.user), pk=...)` answers 404.
    """
    if not _signed_in(actor):
        return Event.objects.none()
    live = Event.objects.filter(archived_at__isnull=True)
    if actor.role in _ADMINS:
        return live
    scope = Q(extension__isnull=True)
    if actor.extension_id is not None:
        scope |= Q(extension_id=actor.extension_id)
    return live.filter(scope)


def can_view(actor, event):
    """May `actor` see this event? Same rule as `visible_events`."""
    if not _signed_in(actor) or event.archived_at is not None:
        return False
    if event.extension_id is None or actor.role in _ADMINS:
        return True
    return _own_extension(actor, event.extension_id)


# ----------------------------------------------------------------- managing
def can_create(actor, extension_id):
    """May `actor` create an event for this extension (None = a global event)?"""
    if not _signed_in(actor):
        return False
    if actor.role in _ADMINS:
        return True
    if actor.role == Role.COORDINATOR:
        return _own_extension(actor, extension_id)
    return False


def can_manage(actor, event):
    """Role and scope only: may `actor` change or cancel this event (whatever its state)?"""
    if not can_view(actor, event):
        return False
    if actor.role in _ADMINS:
        return True
    if actor.role == Role.COORDINATOR:
        return _own_extension(actor, event.extension_id)
    return False


def can_edit(actor, event):
    """May `actor` edit this event right now? Managers only, and never a cancelled event."""
    return not event.is_cancelled and can_manage(actor, event)


def can_cancel(actor, event):
    """May `actor` cancel this event right now? Managers only, and only once."""
    return not event.is_cancelled and can_manage(actor, event)


def available_actions(actor, event):
    """Actions to OFFER on an event (chips, detail page, list rows). The services check again."""
    allowed = set()
    if can_view(actor, event):
        allowed.add(VIEW)
    if can_edit(actor, event):
        allowed.add(EDIT)
    if can_cancel(actor, event):
        allowed.add(CANCEL)
    return allowed


# ----------------------------------------------------------------- page-level
def can_create_global(actor):
    """May `actor` create events that apply to every extension?"""
    return can_create(actor, None)


def can_manage_events(actor):
    """May `actor` see New event at all (admin, super-admin, or a coordinator with an extension)?"""
    if not _signed_in(actor):
        return False
    if actor.role in _ADMINS:
        return True
    return actor.role == Role.COORDINATOR and actor.extension_id is not None


def creatable_extensions(actor):
    """Extensions `actor` may create an event for, as a queryset (the form's choices).

    The QUERYSET form of `can_create`. Admin and super-admin: every extension that is not
    archived (plus "global", which the form offers when `can_create_global`). Coordinator: their
    own. Anyone else: none.
    """
    if not _signed_in(actor):
        return Extension.objects.none()
    live = Extension.objects.filter(archived_at__isnull=True)
    if actor.role in _ADMINS:
        return live
    if actor.role == Role.COORDINATOR and actor.extension_id is not None:
        return live.filter(pk=actor.extension_id)
    return Extension.objects.none()


def require(check, actor, obj=None):
    """Raise PermissionDenied unless `check(actor, obj)` is true; services call this first.

    `check` is one of the functions above, e.g. `require(can_manage, actor, event)`. For
    `can_create` pass the extension id (or None) as the third argument.
    """
    if not check(actor, obj):
        raise PermissionDenied("You cannot do this with this event.")
